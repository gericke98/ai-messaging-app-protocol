    ```rust
    async fn handle_evict(
        ws_sender: &mut WsSender,
        identity: &Identity,
        thread_id: &str,

        let mut intro = serde_json::json!({

            "timestamp": protocol::now_rfc3339(),

                    "text": "Demonstrating AINM memory eviction: removing a previously stored archival fact.",

                }
            ]
        });
        signing::sign_envelope(&identity.key_pair, &mut intro)?;
        send_message(ws_sender, &intro).await?;

        tokio::time::sleep(std::time::Duration::from_millis(300)).await;

        let evict_req = protocol::new_request(
            "ainm.memory.evict",
            serde_json::json!({

                "tier": "archival",
                "key": "user_preference.demo"

            serde_json::json!(protocol::new_message_id()),
        );
        let evict_text = serde_json::to_string(&evict_req)?;
        ws_sender
            .send(Message::Text(evict_text.into()))
            .await
            .map_err(|e| anyhow::anyhow!("send memory evict: {}", e))?;

        Ok(())
    }
    ```

    ```rust
    /// Agent Card HTTP endpoint with ETag caching (per D-03, D-04).
    async fn agent_card_handler(
        headers: HeaderMap,
        State(state): State<ServerState>,
    ) -> Response {
        let body = serde_json::to_vec(&state.agent_card).unwrap();
        let hash = ring::digest::digest(&ring::digest::SHA256, &body);
        let etag_value = format!(
            "\"{}\"",
            base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(hash.as_ref())
        );

        // Check If-None-Match (per D-04)
        if let Some(inm) = headers.get(header::IF_NONE_MATCH) {
            if let Ok(inm_str) = inm.to_str() {
                if inm_str == etag_value {
                    return (
                        StatusCode::NOT_MODIFIED,
                        [(header::ETAG, etag_value)],
                    )
                        .into_response();
                }
            }
        }

        // Return 200 with body and ETag
        (
            StatusCode::OK,
            [
                (header::CONTENT_TYPE, "application/json".to_string()),
                (header::ETAG, etag_value),
            ],
            body,
        )
            .into_response()
    }
    ```
