       ```swift
       import GRDB

       /// Data transfer object for conversation list rows.
       struct ConversationRow: Codable, Sendable, FetchableRecord {
           let id: String
           var title: String?
           let createdAt: String
           let createdBy: String
           var hasUnread: Bool
           var lastMessageTimestamp: String?
           var lastMessageContent: String?  // preview text from contentJson
           var lastMessageFrom: String?
       }

       struct ThreadRepository: Sendable {
           let dbPool: DatabasePool

           /// Returns an AsyncValueObservation of conversation rows sorted by most recent message.
           func observeConversations() -> ValueObservation<[ConversationRow]> {
               ValueObservation.tracking { db in
                   try Row.fetchAll(db, sql: """
                       SELECT t.*,
                              m.timestamp AS lastMessageTimestamp,
                              m.contentJson AS lastMessageContent,
                              m.fromAddress AS lastMessageFrom
                       FROM thread t
                       LEFT JOIN message m ON m.id = (
                           SELECT m2.id FROM message m2
                           WHERE m2.threadId = t.id AND m2.type = 'message'
                           ORDER BY m2.timestamp DESC LIMIT 1
                       )
                       ORDER BY COALESCE(m.timestamp, t.createdAt) DESC
                       """)
                   .map { row in
                       ConversationRow(
                           id: row["id"],
                           title: row["title"],
                           createdAt: row["createdAt"],
                           createdBy: row["createdBy"],
                           hasUnread: row["hasUnread"],
                           lastMessageTimestamp: row["lastMessageTimestamp"],
                           lastMessageContent: row["lastMessageContent"],
                           lastMessageFrom: row["lastMessageFrom"]
                       )
                   }
               }
           }

           /// Mark a thread as read.
           func markAsRead(threadId: String) async throws {
               try await dbPool.write { db in
                   try db.execute(sql: "UPDATE thread SET hasUnread = 0 WHERE id = ?", arguments: [threadId])
               }
           }

           /// Mark a thread as unread.
           func markAsUnread(threadId: String) async throws {
               try await dbPool.write { db in
                   try db.execute(sql: "UPDATE thread SET hasUnread = 1 WHERE id = ?", arguments: [threadId])
               }
           }
       }
       ```

    4. Create NEW file `MessageRepository.swift` with:
       ```swift
       import GRDB
       import Foundation

       struct MessageRepository: Sendable {
           let dbPool: DatabasePool
           private let pageSize = 50

           /// Fetch a page of messages for a thread, ordered by timestamp descending.
           /// Returns messages in chronological order (reversed from query).
           func fetchMessages(threadId: String, olderThan: String? = nil) async throws -> [MessageRecord] {
               try await dbPool.read { db in
                   var sql = "SELECT * FROM message WHERE threadId = ? AND type = 'message'"
                   var arguments: [DatabaseValueConvertible] = [threadId]
                   if let olderThan {
                       sql += " AND timestamp < ?"
                       arguments.append(olderThan)
                   }
                   sql += " ORDER BY timestamp DESC LIMIT ?"
                   arguments.append(pageSize)
                   return try MessageRecord.fetchAll(db, sql: sql, arguments: StatementArguments(arguments))
               }.reversed()  // Return in chronological order
           }

           /// Observe new messages arriving for a specific thread (messages newer than a given timestamp).
           func observeNewMessages(threadId: String, newerThan: String) -> ValueObservation<[MessageRecord]> {
               ValueObservation.tracking { db in
                   try MessageRecord
                       .filter(Column("threadId") == threadId)
                       .filter(Column("type") == "message")
                       .filter(Column("timestamp") > newerThan)
                       .order(Column("timestamp").asc)
                       .fetchAll(db)
               }
           }

           /// Fetch participants for a thread.
           func fetchParticipants(threadId: String) async throws -> [ParticipantRecord] {
               try await dbPool.read { db in
                   try ParticipantRecord
                       .filter(Column("threadId") == threadId)
                       .fetchAll(db)
               }
           }
       }
       ```
