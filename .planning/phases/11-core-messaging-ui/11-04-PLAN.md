       ```swift
       import Foundation
       import Observation
       import GRDB

       /// Group of consecutive messages from the same sender per D-03.
       struct MessageGroup: Identifiable {
           let id: String       // first message ID
           let senderAddress: String
           let senderType: String  // "human" or "agent"
           let senderName: String  // display name portion of AINM address
           let messages: [MessageRecord]
           let isFromCurrentUser: Bool
       }

       @MainActor
       @Observable
       final class MessageThreadViewModel {
           var messageGroups: [MessageGroup] = []
           var isLoadingMore = false
           var hasMoreMessages = true
           var threadTitle: String = ""
           var participantCount: Int = 0

           private let threadId: String
           private let currentUserIdentity: String
           private let repository: MessageRepository
           private let threadRepository: ThreadRepository
           private var participants: [String: ParticipantRecord] = [:]  // identity -> record
           private var allMessages: [MessageRecord] = []
           private var oldestTimestamp: String?
           private var newestTimestamp: String?
           private var observationTask: Task<Void, Never>?

           init(threadId: String, currentUserIdentity: String, database: AppDatabase) {
               self.threadId = threadId
               self.currentUserIdentity = currentUserIdentity
               self.repository = MessageRepository(dbPool: database.dbPool)
               self.threadRepository = ThreadRepository(dbPool: database.dbPool)
           }

           func loadInitial() async {
               // Mark thread as read
               try? await threadRepository.markAsRead(threadId: threadId)

               // Load participants
               let parts = (try? await repository.fetchParticipants(threadId: threadId)) ?? []
               for p in parts { participants[p.identity] = p }
               participantCount = parts.count

               // Load thread title
               // (fetch from GRDB directly)
               if let thread = try? await threadRepository.dbPool.read({ db in
                   try ThreadRecord.fetchOne(db, key: threadId)
               }) {
                   threadTitle = thread.title ?? "Conversation"
               }

               // Load initial page
               let messages = (try? await repository.fetchMessages(threadId: threadId)) ?? []
               self.allMessages = Array(messages)
               self.oldestTimestamp = allMessages.first?.timestamp
               self.newestTimestamp = allMessages.last?.timestamp
               self.hasMoreMessages = messages.count == 50
               rebuildGroups()

               // Start observing new messages
               startObservingNewMessages()
           }

           func loadOlderMessages() async {
               guard !isLoadingMore, hasMoreMessages, let oldest = oldestTimestamp else { return }
               isLoadingMore = true
               let older = (try? await repository.fetchMessages(threadId: threadId, olderThan: oldest)) ?? []
               if !older.isEmpty {
                   allMessages.insert(contentsOf: older, at: 0)
                   oldestTimestamp = allMessages.first?.timestamp
               }
               hasMoreMessages = older.count == 50
               isLoadingMore = false
               rebuildGroups()
           }

           func stopObserving() {
               observationTask?.cancel()
               observationTask = nil
           }

           private func startObservingNewMessages() {
               guard let newest = newestTimestamp else { return }
               observationTask = Task {
                   let observation = repository.observeNewMessages(threadId: threadId, newerThan: newest)
                   do {
                       for try await newMessages in observation.values(in: repository.dbPool) {
                           // Append only messages not already in allMessages
                           let existingIds = Set(allMessages.map(\.id))
                           let truly_new = newMessages.filter { !existingIds.contains($0.id) }
                           if !truly_new.isEmpty {
                               allMessages.append(contentsOf: truly_new)
                               newestTimestamp = allMessages.last?.timestamp
                               rebuildGroups()
                           }
                       }
                   } catch {
                       // Observation ended
                   }
               }
           }

           private func rebuildGroups() {
               var groups: [MessageGroup] = []
               var current: [MessageRecord] = []
               var currentSender: String?

               for message in allMessages {
                   if message.fromAddress == currentSender {
                       current.append(message)
                   } else {
                       if !current.isEmpty, let sender = currentSender {
                           groups.append(makeGroup(messages: current, sender: sender))
                       }
                       current = [message]
                       currentSender = message.fromAddress
                   }
               }
               if !current.isEmpty, let sender = currentSender {
                   groups.append(makeGroup(messages: current, sender: sender))
               }
               messageGroups = groups
           }

           private func makeGroup(messages: [MessageRecord], sender: String) -> MessageGroup {
               let participant = participants[sender]
               let senderType = participant?.type ?? "human"
               // Extract display name from AINM address (part before @)
               let senderName = sender.components(separatedBy: "@").first ?? sender
               return MessageGroup(
                   id: messages[0].id,
                   senderAddress: sender,
                   senderType: senderType,
                   senderName: senderName,
                   messages: messages,
                   isFromCurrentUser: sender == currentUserIdentity
               )
           }

           /// Decode content blocks from a MessageRecord's contentJson field.
           func contentBlocks(for message: MessageRecord) -> [ContentBlock] {
               guard let data = message.contentJson.data(using: .utf8),
                     let blocks = try? JSONDecoder().decode([ContentBlock].self, from: data) else {
                   return [.unknown(fallback: "[Could not decode message content]")]
               }
               return blocks
           }

           /// Format timestamp for display.
           func formatTimestamp(_ timestamp: String) -> String {
               guard let date = ISO8601DateFormatter.ainm.date(from: timestamp) else { return "" }
               let formatter = RelativeDateTimeFormatter()
               formatter.unitsStyle = .abbreviated
               return formatter.localizedString(for: date, relativeTo: Date())
           }
       }
       ```

       ```swift
       import SwiftUI

       /// Renders a group of consecutive messages from the same sender.
       /// Per D-03: avatar+name on first message only, 4pt spacing within group, 16pt between groups.
       struct MessageGroupView: View {
           let group: MessageGroup
           let contentBlocksProvider: (MessageRecord) -> [ContentBlock]
           let timestampFormatter: (String) -> String

           var body: some View {
               VStack(alignment: group.isFromCurrentUser ? .trailing : .leading, spacing: 4) {
                   ForEach(Array(group.messages.enumerated()), id: \.element.id) { index, message in
                       let blocks = contentBlocksProvider(message)
                       let isFirst = index == 0
                       let isLast = index == group.messages.count - 1

                       if group.isFromCurrentUser {
                           MessageBubbleView(
                               blocks: blocks,
                               senderName: group.senderName,
                               showHeader: isFirst,
                               timestamp: isLast ? timestampFormatter(message.timestamp) : nil
                           )
                       } else {
                           AgentMessageView(
                               blocks: blocks,
                               senderName: group.senderName,
                               senderType: group.senderType,
                               showHeader: isFirst,
                               timestamp: isLast ? timestampFormatter(message.timestamp) : nil
                           )
                       }
                   }
               }
               .padding(.horizontal, 16)
           }
       }
       ```

       ```swift
       import SwiftUI

       struct MessageThreadView: View {
           @Bindable var appState: AppState
           let threadId: String
           @State private var viewModel: MessageThreadViewModel?
           @State private var scrollPosition = ScrollPosition(idType: String.self)

           var body: some View {
               VStack(spacing: 0) {
                   ScrollView {
                       LazyVStack(spacing: 16) {
                           if let viewModel {
                               if viewModel.isLoadingMore {
                                   ProgressView("Loading older messages...")
                                       .padding()
                               }

                               ForEach(viewModel.messageGroups) { group in
                                   MessageGroupView(
                                       group: group,
                                       contentBlocksProvider: { viewModel.contentBlocks(for: $0) },
                                       timestampFormatter: { viewModel.formatTimestamp($0) }
                                   )
                               }
                           }
                       }
                   }
                   .scrollPosition($scrollPosition)
                   .defaultScrollAnchor(.bottom)
                   .onScrollGeometryChange(for: Bool.self) { geometry in
                       geometry.contentOffset.y <= 20
                   } action: { _, isNearTop in
                       if isNearTop {
                           Task { await viewModel?.loadOlderMessages() }
                       }
                   }

                   ComposerView(
                       isOffline: appState.connectionState != .connected,
                       onSend: { text in
                           Task { await sendMessage(text: text) }
                       }
                   )
               }
               .navigationTitle(viewModel?.threadTitle ?? "Conversation")
               .toolbar {
                   ToolbarItem(placement: .principal) {
                       VStack(spacing: 2) {
                           Text(viewModel?.threadTitle ?? "Conversation")
                               .font(.subheadline.bold())
                           if let count = viewModel?.participantCount, count > 0 {
                               Text("\(count) participants")
                                   .font(.caption)
                                   .foregroundStyle(.secondary)
                           }
                       }
                   }
               }
               .onAppear {
                   appState.currentViewingThreadId = threadId
                   if viewModel == nil, let db = appState.appDatabase {
                       let vm = MessageThreadViewModel(
                           threadId: threadId,
                           currentUserIdentity: appState.connectedIdentity ?? "",
                           database: db
                       )
                       self.viewModel = vm
                       Task { await vm.loadInitial() }
                   }
               }
               .onDisappear {
                   appState.currentViewingThreadId = nil
                   viewModel?.stopObserving()
               }
           }

           private func sendMessage(text: String) async {
               guard let identity = appState.connectedIdentity else { return }
               let envelope = MessageEnvelope(
                   id: UUID().uuidString,
                   from: identity,
                   threadId: threadId,
                   timestamp: ISO8601DateFormatter.ainm.string(from: Date()),
                   type: .message,
                   content: [.text(ContentBlock.TextContent(
                       version: "1.0",
                       source: "user",
                       text: text,
                       mentions: nil,
                       fallback: text
                   ))],
                   signature: nil,
                   parentId: nil
               )
               // Access client through appState -- need to add a sendMessage method to AppState
               try? await appState.sendMessage(envelope: envelope)
           }
       }
       ```

       ```swift
       /// Create a new thread and send the first message.
       /// Per threading.md: sends thread.create envelope, then the first message envelope.
       /// Both are persisted to GRDB via write-through.
       func createThread(title: String?, firstMessageText: String, identity: String) async throws -> String {
           guard case .connected = state else { throw ClientError.notConnected }

           let threadId = UUID().uuidString
           let timestamp = ISO8601DateFormatter.ainm.string(from: Date())

           // 1. Persist thread locally first (write-through)
           let threadRecord = ThreadRecord(
               id: threadId,
               title: title,
               createdAt: timestamp,
               createdBy: identity,
               parentThreadId: nil,
               parentMessageId: nil
           )
           try await appDatabase.dbPool.write { db in
               try threadRecord.save(db)
           }

           // 2. Send thread.create via protocol
           let createEnvelope = MessageEnvelope(
               id: UUID().uuidString,
               from: identity,
               threadId: threadId,
               timestamp: timestamp,
               type: .threadCreate,
               content: [],
               signature: nil,
               parentId: nil
           )
           try await sendMessage(envelope: createEnvelope)

           // 3. Send first message
           let messageEnvelope = MessageEnvelope(
               id: UUID().uuidString,
               from: identity,
               threadId: threadId,
               timestamp: ISO8601DateFormatter.ainm.string(from: Date()),
               type: .message,
               content: [.text(ContentBlock.TextContent(
                   version: "1.0",
                   source: "user",
                   text: firstMessageText,
                   mentions: nil,
                   fallback: firstMessageText
               ))],
               signature: nil,
               parentId: nil
           )
           try await sendMessage(envelope: messageEnvelope)

           return threadId
       }
       ```
