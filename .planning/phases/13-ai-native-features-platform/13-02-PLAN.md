1. Create `UI/Mentions/MentionTextRenderer.swift` with a single public function:
   ```swift
   import SwiftUI

   enum MentionTextRenderer {
       /// Build an AttributedString from text content with mention highlights.
       /// Per UI-SPEC: accent blue bold for agent/default, white bold in human bubbles.
       static func highlightMentions(
           in attributed: AttributedString,
           mentions: [ContentBlock.Mention]?,
           isHumanBubble: Bool
       ) -> AttributedString {
           guard let mentions = mentions, !mentions.isEmpty else { return attributed }
           var result = attributed

           // Sort by offset descending to avoid index shifting
           for mention in mentions.sorted(by: { $0.offset > $1.offset }) {
               // Bounds check
               let startOffset = mention.offset
               let endOffset = mention.offset + mention.length
               guard startOffset >= 0,
                     endOffset <= result.characters.count else { continue }

               let startIndex = result.characters.index(
                   result.characters.startIndex, offsetBy: startOffset
               )
               let endIndex = result.characters.index(
                   result.characters.startIndex, offsetBy: endOffset
               )
               let range = startIndex..<endIndex

               // Apply highlight style per UI-SPEC
               result[range].foregroundColor = isHumanBubble ? .white : .accentColor
               result[range].font = .body.bold()
           }
           return result
       }
   }
   ```

1. Create `UI/Mentions/MentionAutocompleteView.swift`:
   ```swift
   import SwiftUI

   /// Participant data for mention autocomplete.
   struct MentionCandidate: Identifiable {
       let id: String  // identity (AINM address)
       let displayName: String
       let type: String  // "human" or "agent"
   }

   /// Overlay popup for @-mention autocomplete per UI-SPEC Component 2.
   struct MentionAutocompleteView: View {
       let candidates: [MentionCandidate]
       let filter: String  // text after @
       var onSelect: (MentionCandidate) -> Void

       private var filtered: [MentionCandidate] {
           if filter.isEmpty { return candidates }
           return candidates.filter {
               $0.displayName.localizedCaseInsensitiveContains(filter)
           }
       }

       var body: some View {
           VStack(spacing: 0) {
               if filtered.isEmpty {
                   Text("No matches")
                       .font(.subheadline)
                       .foregroundStyle(Color(.secondaryLabel))
                       .frame(height: 48)
                       .frame(maxWidth: .infinity)
               } else {
                   ScrollView {
                       LazyVStack(spacing: 0) {
                           ForEach(filtered) { candidate in
                               Button { onSelect(candidate) } label: {
                                   autocompleteRow(candidate)
                               }
                               .buttonStyle(.plain)
                               if candidate.id != filtered.last?.id {
                                   Divider()
                                       .foregroundStyle(Color(.separator))
                               }
                           }
                       }
                   }
                   .frame(maxHeight: 200)
               }
           }
           .background(Color(.systemBackground))
           .clipShape(RoundedRectangle(cornerRadius: 12))
           .shadow(color: .black.opacity(0.15), radius: 8, y: -4)
           .overlay(
               RoundedRectangle(cornerRadius: 12)
                   .stroke(Color(.separator), lineWidth: 0.5)
           )
           .accessibilityLabel("Mention suggestions")
       }

       private func autocompleteRow(_ candidate: MentionCandidate) -> some View {
           HStack(spacing: 8) {
               AvatarView(participantType: candidate.type, size: 24)
               VStack(alignment: .leading, spacing: 0) {
                   Text(candidate.displayName)
                       .font(.body)
                       .foregroundStyle(Color(.label))
                       .lineLimit(1)
                   Text(candidate.id)
                       .font(.caption)
                       .foregroundStyle(Color(.secondaryLabel))
                       .lineLimit(1)
               }
               Spacer()
           }
           .padding(.horizontal, 12)
           .frame(height: 48)
           .contentShape(Rectangle())
           .accessibilityLabel("\(candidate.displayName), \(candidate.type)")
       }
   }
   ```
