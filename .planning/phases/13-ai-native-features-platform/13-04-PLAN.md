   import Foundation

   struct ContextFrame: Codable, Sendable {
       let threadId: String
       let frame: Frame

       struct Frame: Codable, Sendable {
           let activeContext: ActiveContext
           let conversationSummary: String?
           let archivalFacts: [ArchivalFact]
           let participants: [FrameParticipant]

           enum CodingKeys: String, CodingKey {
               case activeContext = "active_context"
               case conversationSummary = "conversation_summary"
               case archivalFacts = "archival_facts"
               case participants
           }
       }

       struct ActiveContext: Codable, Sendable {
           let recentMessages: [MessageSummary]
           let windowStart: String
           let messageCount: Int

           enum CodingKeys: String, CodingKey {
               case recentMessages = "recent_messages"
               case windowStart = "window_start"
               case messageCount = "message_count"
           }
       }

       struct MessageSummary: Codable, Sendable, Identifiable {
           let id: String
           let from: String
           let timestamp: String
           let summary: String
       }

       struct ArchivalFact: Codable, Sendable, Identifiable {
           let factId: String
           let key: String
           let value: AnyCodableValue
           let tags: [String]?
           let storedBy: String
           let storedAt: String

           var id: String { factId }

           enum CodingKeys: String, CodingKey {
               case factId = "fact_id"
               case key, value, tags
               case storedBy = "stored_by"
               case storedAt = "stored_at"
           }
       }

       struct FrameParticipant: Codable, Sendable {
           let identity: String
           let type: String
           let presence: String
       }

       enum CodingKeys: String, CodingKey {
           case threadId = "thread_id"
           case frame
       }
   }

   /// Response from ainm.memory.query
   struct MemoryQueryResult: Codable, Sendable {
       let facts: [ContextFrame.ArchivalFact]
   }

2. Create `Persistence/MemoryFactRecord.swift`:
   ```swift
   import GRDB
   import Foundation

   struct MemoryFactRecord: Codable, Sendable, FetchableRecord, PersistableRecord {
       static let databaseTableName = "memoryFact"

       let factId: String
       let key: String
       let value: String       // JSON string
       let tags: String?       // JSON array string
       let storedBy: String
       let storedAt: String

       /// Create from ArchivalFact.
       init(from fact: ContextFrame.ArchivalFact) {
           self.factId = fact.factId
           self.key = fact.key
           self.value = (try? String(data: JSONEncoder().encode(fact.value), encoding: .utf8)) ?? ""
           if let tags = fact.tags {
               self.tags = (try? String(data: JSONEncoder().encode(tags), encoding: .utf8))
           } else {
               self.tags = nil
           }
           self.storedBy = fact.storedBy
           self.storedAt = fact.storedAt
       }

       /// Memberwise.
       init(factId: String, key: String, value: String, tags: String?, storedBy: String, storedAt: String) {
           self.factId = factId
           self.key = key
           self.value = value
           self.tags = tags
           self.storedBy = storedBy
           self.storedAt = storedAt
       }

       /// Decode tags from JSON string.
       func decodedTags() -> [String]? {
           guard let tags, let data = tags.data(using: .utf8) else { return nil }
           return try? JSONDecoder().decode([String].self, from: data)
       }
   }
   ```
