   struct AgentCardRecord: Codable, Sendable, FetchableRecord, PersistableRecord {
       static let databaseTableName = "agentCard"

       let url: String          // Primary key — agent card URL
       let identity: String     // AINM address
       let cardJson: String     // Full JSON
       var etag: String?        // HTTP ETag
       let fetchedAt: String    // RFC 3339

       /// Decode the cached AgentCard from JSON.
       func agentCard() -> AgentCard? {
           guard let data = cardJson.data(using: .utf8) else { return nil }
           return try? JSONDecoder().decode(AgentCard.self, from: data)
       }
   }

4. Create `Networking/AgentCardService.swift`:
   ```swift
   import Foundation
   import GRDB

   /// Fetches and caches Agent Cards via HTTP with ETag-based conditional requests.
   actor AgentCardService {
       private let appDatabase: AppDatabase
       private let session = URLSession.shared

       init(appDatabase: AppDatabase) {
           self.appDatabase = appDatabase
       }

       /// Fetch Agent Card from URL with ETag-based conditional request.
       /// Returns cached version if server returns 304 Not Modified.
       func fetchCard(agentCardURL: String, identity: String) async throws -> AgentCard {
           guard let url = URL(string: agentCardURL) else {
               throw AgentCardError.invalidURL
           }

           // Check cache for ETag
           let cachedRecord = try await appDatabase.dbPool.read { db in
               try AgentCardRecord.fetchOne(db, key: agentCardURL)
           }

           var request = URLRequest(url: url)
           request.setValue("application/json", forHTTPHeaderField: "Accept")
           if let etag = cachedRecord?.etag {
               request.setValue(etag, forHTTPHeaderField: "If-None-Match")
           }

           let (data, response) = try await session.data(for: request)
           guard let httpResponse = response as? HTTPURLResponse else {
               throw AgentCardError.invalidResponse
           }

           if httpResponse.statusCode == 304, let cached = cachedRecord?.agentCard() {
               return cached
           }

           guard httpResponse.statusCode == 200 else {
               throw AgentCardError.httpError(httpResponse.statusCode)
           }

           let card = try JSONDecoder().decode(AgentCard.self, from: data)
           let etag = httpResponse.value(forHTTPHeaderField: "ETag")
           let record = AgentCardRecord(
               url: agentCardURL,
               identity: identity,
               cardJson: String(data: data, encoding: .utf8) ?? "",
               etag: etag,
               fetchedAt: ISO8601DateFormatter.ainm.string(from: Date())
           )
           try await appDatabase.dbPool.write { db in
               try record.save(db)
           }
           return card
       }

       /// Load all cached agent cards from GRDB.
       func loadCachedCards() async throws -> [(identity: String, card: AgentCard)] {
           let records = try await appDatabase.dbPool.read { db in
               try AgentCardRecord.fetchAll(db)
           }
           return records.compactMap { record in
               guard let card = record.agentCard() else { return nil }
               return (identity: record.identity, card: card)
           }
       }

       /// Fetch Agent Card by AINM address using identity resolution.
       /// For v1: constructs the well-known URL from the address domain.
       func fetchCardByAddress(_ address: String) async throws -> AgentCard {
           // Extract domain from address (e.g., "agent@server.com" -> "server.com")
           let parts = address.components(separatedBy: "@")
           guard parts.count == 2, let domain = parts.last, !domain.isEmpty else {
               throw AgentCardError.invalidAddress
           }
           let localPart = parts[0]
           // Well-known URL pattern: https://{domain}/.well-known/ainm/agents/{local}/agent-card.json
           let cardURL = "https://\(domain)/.well-known/ainm/agents/\(localPart)/agent-card.json"
           return try await fetchCard(agentCardURL: cardURL, identity: address)
       }

       enum AgentCardError: Error, LocalizedError {
           case invalidURL
           case invalidResponse
           case invalidAddress
           case httpError(Int)

           var errorDescription: String? {
               switch self {
               case .invalidURL: return "Invalid agent card URL"
               case .invalidResponse: return "Invalid server response"
               case .invalidAddress: return "Invalid AINM address format"
               case .httpError(let code): return "Server returned HTTP \(code)"
               }
           }
       }
   }
   ```

1. Create `UI/AgentDiscovery/AgentDiscoveryView.swift` per UI-SPEC Component 4:
   ```swift
   import SwiftUI

   struct AgentDiscoveryView: View {
       let appState: AppState

       @State private var agents: [(identity: String, card: AgentCard)] = []
       @State private var searchText = ""
       @State private var isLoading = true
       @State private var addAgentAddress = ""
       @State private var addAgentError: String?
       @State private var isAddingAgent = false

       private var filteredAgents: [(identity: String, card: AgentCard)] {
           if searchText.isEmpty { return agents }
           return agents.filter { item in
               item.card.name.localizedCaseInsensitiveContains(searchText) ||
               item.card.description.localizedCaseInsensitiveContains(searchText) ||
               (item.card.skills.flatMap { $0.tags ?? [] }.contains { $0.localizedCaseInsensitiveContains(searchText) })
           }
       }

       var body: some View {
           Group {
               if isLoading {
                   ProgressView()
                       .frame(maxWidth: .infinity, maxHeight: .infinity)
               } else if agents.isEmpty {
                   emptyState
               } else {
                   agentList
               }
           }
           .navigationTitle("Discover Agents")
           .searchable(text: $searchText, prompt: "Search agents...")
           .refreshable { await loadAgents() }
           .task { await loadAgents() }
       }

       private var emptyState: some View {
           VStack(spacing: 16) {
               Image(systemName: "magnifyingglass")
                   .font(.system(size: 48))
                   .foregroundStyle(Color(.tertiaryLabel))
               Text("No agents discovered")
                   .font(.title2.bold())
               Text("Connect to an agent by entering their AINM address")
                   .font(.subheadline)
                   .foregroundStyle(Color(.secondaryLabel))
               addAgentField
           }
           .padding(.horizontal, 32)
       }

       private var addAgentField: some View {
           VStack(spacing: 8) {
               TextField("agent@server.example", text: $addAgentAddress)
                   .textFieldStyle(.roundedBorder)
                   .autocapitalization(.none)
                   .keyboardType(.emailAddress)
               if let error = addAgentError {
                   Text(error)
                       .font(.caption)
                       .foregroundStyle(Color.red)
               }
               Button("Add Agent") { Task { await addAgent() } }
                   .buttonStyle(.borderedProminent)
                   .disabled(addAgentAddress.isEmpty || isAddingAgent)
           }
       }

       private var agentList: some View {
           List {
               ForEach(filteredAgents, id: \.identity) { item in
                   NavigationLink {
                       AgentCardDetailView(
                           identity: item.identity,
                           card: item.card,
                           appState: appState
                       )
                   } label: {
                       agentRow(item.card)
                   }
               }
           }
           .listStyle(.plain)
       }
   ```
