3. **Add sub-thread creation to AINMClient.swift:**

   ```swift
   func createSubThread(
       parentThreadId: String,
       parentMessageId: String,
       identity: String
   ) async throws -> String {
       guard case .connected = state else { throw ClientError.notConnected }

       let threadId = UUID().uuidString
       let timestamp = ISO8601DateFormatter.ainm.string(from: Date())

       // Persist sub-thread locally
       let threadRecord = ThreadRecord(
           id: threadId,
           title: nil,
           createdAt: timestamp,
           createdBy: identity,
           parentThreadId: parentThreadId,
           parentMessageId: parentMessageId
       )
       try await appDatabase.dbPool.write { db in
           try threadRecord.save(db)
       }

       // Send thread.create with parent fields
       let createEnvelope = MessageEnvelope(
           id: UUID().uuidString,
           from: identity,
           threadId: threadId,
           timestamp: timestamp,
           type: .threadCreate,
           content: [],
           signature: nil,
           parentId: parentMessageId
       )
       try await sendMessage(envelope: createEnvelope)
       return threadId
   }
   ```

1. **Create `App/DeepLinkHandler.swift`** per UI-SPEC Component 10:
   ```swift
   import Foundation

   enum DeepLinkHandler {
       enum Route {
           case thread(threadId: String)
           case message(threadId: String, messageId: String)
       }

       /// Parse an ainm:// URL into a navigation route.
       /// Supported:
       ///   ainm://thread/{thread_id}
       ///   ainm://thread/{thread_id}/message/{message_id}
       static func parse(_ url: URL) -> Route? {
           guard url.scheme == "ainm" else { return nil }

           let urlString = url.absoluteString
           guard urlString.hasPrefix("ainm://thread/") else { return nil }
           let remainder = String(urlString.dropFirst("ainm://thread/".count))
           let parts = remainder.components(separatedBy: "/")

           guard let threadId = parts.first, !threadId.isEmpty else { return nil }

           if parts.count >= 3, parts[1] == "message", !parts[2].isEmpty {
               return .message(threadId: threadId, messageId: parts[2])
           }

           return .thread(threadId: threadId)
       }
   }
   ```
