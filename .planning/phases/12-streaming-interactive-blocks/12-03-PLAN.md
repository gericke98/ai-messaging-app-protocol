**StreamingViewModel.swift:**
```swift
@MainActor
@Observable
final class StreamingViewModel {
    struct StreamingMessageState: Identifiable, Sendable {
        let id: String  // streamId
        let messageId: String
        let from: String
        let threadId: String
        var blocks: [StreamingBlockState]
        var status: StreamStatus = .streaming
        var finalizingStartTime: Date? = nil

        enum StreamStatus: Sendable {
            case streaming
            case cancelling
            case finalizing
            case notVerified
        }
    }

    struct StreamingBlockState: Sendable {
        let type: String   // "text" or "code"
        var content: String = ""
        var language: String?
    }

    var activeStreams: [String: StreamingMessageState] = [:]  // streamId -> state

    func streamStarted(_ stream: StreamingManager.ActiveStream) {
        let blocks = stream.blockLayout.map { layout in
            StreamingBlockState(type: layout.type, content: "", language: nil)
        }
        let state = StreamingMessageState(
            id: stream.streamId,
            messageId: stream.messageId,
            from: stream.from,
            threadId: stream.threadId,
            blocks: blocks
        )
        activeStreams[stream.streamId] = state
    }

    func updateStreamContent(streamId: String, blockIndex: Int, content: String) {
        guard var stream = activeStreams[streamId],
              blockIndex >= 0 && blockIndex < stream.blocks.count else { return }
        stream.blocks[blockIndex].content = content
        activeStreams[streamId] = stream
    }

    func updateBlockLanguage(streamId: String, blockIndex: Int, language: String) {
        guard var stream = activeStreams[streamId],
              blockIndex >= 0 && blockIndex < stream.blocks.count else { return }
        stream.blocks[blockIndex].language = language
        activeStreams[streamId] = stream
    }

    func markCancelling(streamId: String) {
        activeStreams[streamId]?.status = .cancelling
    }

    func streamEnded(streamId: String, status: String) {
        guard var stream = activeStreams[streamId] else { return }
        // Transition to finalizing -- keep content visible, remove cursor
        stream.status = .finalizing
        stream.finalizingStartTime = Date()
        activeStreams[streamId] = stream

        // Start 5-second timeout for collapsed message
        let capturedId = streamId
        Task { @MainActor in
            try? await Task.sleep(for: .seconds(5))
            if let s = activeStreams[capturedId], s.status == .finalizing {
                activeStreams[capturedId]?.status = .notVerified
            }
        }
    }

    func removeStream(streamId: String) {
        activeStreams.removeValue(forKey: streamId)
    }

    /// Remove streaming placeholder when the permanent message arrives (matched by messageId).
    func removePlaceholderForMessage(messageId: String) {
        if let key = activeStreams.first(where: { $0.value.messageId == messageId })?.key {
            activeStreams.removeValue(forKey: key)
        }
    }

    func activeStreamsForThread(_ threadId: String) -> [StreamingMessageState] {
        activeStreams.values.filter { $0.threadId == threadId }.sorted { $0.id < $1.id }
    }

    func hasActiveStreams(inThread threadId: String) -> Bool {
        activeStreams.values.contains { $0.threadId == threadId && $0.status == .streaming }
    }
}
```

```swift
struct StreamingMessageView: View {
    let state: StreamingViewModel.StreamingMessageState

    var body: some View {
        // Uses same left-aligned agent message container style
        HStack(alignment: .top, spacing: 8) {
            // Avatar (same AvatarView from Phase 11, agent style)
            AvatarView(name: senderName, isAgent: true)
                .frame(width: 36, height: 36)

            VStack(alignment: .leading, spacing: 4) {
                // Sender name + "Agent" badge
                HStack(spacing: 4) {
                    Text(senderName)
                        .font(.subheadline.bold())
                    Text("Agent")
                        .font(.caption2)
                        .padding(.horizontal, 4)
                        .padding(.vertical, 2)
                        .background(Color(.tertiarySystemFill))
                        .clipShape(RoundedRectangle(cornerRadius: 4))
                }

                // Streaming blocks in vertical stack
                VStack(alignment: .leading, spacing: 8) {
                    ForEach(Array(state.blocks.enumerated()), id: \.offset) { index, block in
                        streamingBlockView(block: block, isLast: index == state.blocks.count - 1)
                    }
                }

                // Not verified badge
                if state.status == .notVerified {
                    HStack(spacing: 4) {
                        Image(systemName: "exclamationmark.triangle.fill")
                            .font(.caption)
                        Text("Not verified")
                            .font(.caption)
                    }
                    .foregroundStyle(.secondary)
                    .accessibilityLabel("Message not verified")
                    .accessibilityHint("The signed message was not received from the server within the expected time.")
                }
            }
        }
        .padding(.horizontal, 16)
        .accessibilityElement(children: .combine)
        .accessibilityLabel(state.status == .streaming ? "Agent is generating a response" : "Agent response")
    }

    private var senderName: String {
        state.from.components(separatedBy: "@").first ?? state.from
    }

    @ViewBuilder
    private func streamingBlockView(block: StreamingViewModel.StreamingBlockState, isLast: Bool) -> some View {
        if block.type == "code" {
            // Code block with language label and monospaced font
            VStack(alignment: .leading, spacing: 4) {
                if let lang = block.language {
                    Text(lang)
                        .font(.caption)
                        .foregroundStyle(.secondary)
                }
                HStack(spacing: 0) {
                    Text(block.content)
                        .font(.system(.body, design: .monospaced))
                    if isLast && showCursor {
                        CursorView(height: 10)
                    }
                }
            }
            .padding(12)
            .frame(maxWidth: .infinity, alignment: .leading)
            .background(Color(.tertiarySystemBackground))
            .clipShape(RoundedRectangle(cornerRadius: 8))
        } else {
            // Text block
            HStack(spacing: 0) {
                Text(block.content)
                    .font(.body)
                if isLast && showCursor {
                    CursorView(height: 12)
                }
            }
        }
    }

    private var showCursor: Bool {
        state.status == .streaming
    }
}
```

```swift
func setup() throws {
    let db = try AppDatabase.makeDefault()
    self.appDatabase = db
    let client = AINMClient(appDatabase: db)
    self.client = client

    // Wire streaming callbacks
    Task {
        await client.setStreamCallbacks(
            onStart: { [weak self] stream in
                Task { @MainActor in
                    self?.streamingViewModel.streamStarted(stream)
                }
            },
            onDelta: { [weak self] streamId, blockIndex, content in
                Task { @MainActor in
                    self?.streamingViewModel.updateStreamContent(streamId: streamId, blockIndex: blockIndex, content: content)
                }
            },
            onEnd: { [weak self] stream, status in
                Task { @MainActor in
                    self?.streamingViewModel.streamEnded(streamId: stream.streamId, status: status)
                }
            },
            onError: { [weak self] stream in
                Task { @MainActor in
                    self?.streamingViewModel.removeStream(streamId: stream.streamId)
                }
            }
        )
    }
}
```
