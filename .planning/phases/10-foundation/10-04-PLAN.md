**WebSocketConnection.swift** -- actor wrapping URLSessionWebSocketTask:

```swift
import Foundation

actor WebSocketConnection {
    enum ConnectionState: Sendable {
        case disconnected
        case connecting
        case connected
        case disconnecting
    }

    enum ConnectionError: Error, Sendable {
        case notConnected
        case invalidURL
        case connectionFailed(Error)
    }

    private var task: URLSessionWebSocketTask?
    private let session: URLSession
    private(set) var state: ConnectionState = .disconnected

    init(session: URLSession = .shared) {
        self.session = session
    }

    /// Connect to WebSocket URL.
    /// The demo server uses /ws path (not the spec's /ainm/v1).
    func connect(to urlString: String) throws {
        guard let url = URL(string: urlString) else {
            throw ConnectionError.invalidURL
        }
        // Append /ws path for demo server if not already present
        let wsURL: URL
        if url.path.isEmpty || url.path == "/" {
            wsURL = url.appendingPathComponent("ws")
        } else {
            wsURL = url
        }
        state = .connecting
        task = session.webSocketTask(with: wsURL)
        task?.resume()
        state = .connected
    }

    /// Send a JSON-RPC message as text.
    func send(_ data: Data) async throws {
        guard let task, state == .connected else {
            throw ConnectionError.notConnected
        }
        let string = String(data: data, encoding: .utf8)!
        try await task.send(.string(string))
    }

    /// Receive messages as an AsyncStream.
    /// URLSessionWebSocketTask uses a pull model -- must call receive() in a loop.
    func receiveMessages() -> AsyncStream<Result<Data, Error>> {
        AsyncStream { [weak self] continuation in
            Task { [weak self] in
                while let self = self, await self.state == .connected {
                    do {
                        guard let task = await self.task else { break }
                        let message = try await task.receive()
                        switch message {
                        case .string(let text):
                            if let data = text.data(using: .utf8) {
                                continuation.yield(.success(data))
                            }
                        case .data(let data):
                            continuation.yield(.success(data))
                        @unknown default:
                            break
                        }
                    } catch {
                        continuation.yield(.failure(error))
                        continuation.finish()
                        break
                    }
                }
                continuation.finish()
            }
        }
    }

    /// Graceful disconnect.
    func disconnect() {
        state = .disconnecting
        task?.cancel(with: .normalClosure, reason: nil)
        task = nil
        state = .disconnected
    }
}
```

**ReconnectionManager.swift** -- exponential backoff per D-06:

```swift
import Foundation

actor ReconnectionManager {
    private let baseDelay: TimeInterval = 1.0    // 1 second
    private let maxDelay: TimeInterval = 30.0    // 30 second cap per D-06
    private let multiplier: Double = 2.0
    private var currentAttempt: Int = 0
    private var isReconnecting: Bool = false

    /// Calculate next delay using exponential backoff: 1s, 2s, 4s, 8s, 16s, 30s, 30s...
    func nextDelay() -> TimeInterval {
        let delay = min(baseDelay * pow(multiplier, Double(currentAttempt)), maxDelay)
        currentAttempt += 1
        return delay
    }

    /// Reset backoff counter (call after successful connection).
    func reset() {
        currentAttempt = 0
        isReconnecting = false
    }

    /// Attempt reconnection with backoff.
    /// Calls the provided closure after waiting the backoff delay.
    /// Per D-07: always performs full re-handshake (stateless).
    func scheduleReconnect(action: @escaping @Sendable () async throws -> Void) async {
        guard !isReconnecting else { return }
        isReconnecting = true

        let delay = nextDelay()
        try? await Task.sleep(nanoseconds: UInt64(delay * 1_000_000_000))

        guard !Task.isCancelled else {
            isReconnecting = false
            return
        }

        do {
            try await action()
            reset()  // Success -- reset backoff
        } catch {
            isReconnecting = false
            // Caller should call scheduleReconnect again
        }
    }
}
```

Key decisions:
- WebSocket connects to `/ws` path by default (matching demo server, per Research open question #3)
- `ws://` allowed for development (demo server may not have TLS). Production will enforce `wss://`.
- ReconnectionManager backoff: 1s, 2s, 4s, 8s, 16s, 30s cap per D-06
- Per D-07: reconnection always triggers full re-handshake (stateless)

**AINMClient.swift** -- the protocol client that orchestrates WebSocket, crypto, and persistence:

```swift
import Foundation
import CryptoKit
import GRDB

actor AINMClient {
    enum ClientState: Sendable {
        case disconnected
        case connecting
        case handshaking
        case connected(ServerCapabilities)
        case error(String)
    }

    enum ClientError: Error, Sendable {
        case notConnected
        case handshakeFailed(String)
        case encodingError
        case signingError(Error)
    }

    private let webSocket: WebSocketConnection
    private let reconnection: ReconnectionManager
    private let keyManager: KeyManager
    private let appDatabase: AppDatabase
    private var nextRequestId: Int = 1

    private(set) var state: ClientState = .disconnected
    private(set) var serverCapabilities: ServerCapabilities?
    private var receiveTask: Task<Void, Never>?

    init(
        webSocket: WebSocketConnection = WebSocketConnection(),
        reconnection: ReconnectionManager = ReconnectionManager(),
        keyManager: KeyManager = KeyManager(),
        appDatabase: AppDatabase
    ) {
        self.webSocket = webSocket
        self.reconnection = reconnection
        self.keyManager = keyManager
        self.appDatabase = appDatabase
    }

    /// Full connection flow: connect WebSocket + handshake.
    /// Per D-07: always full re-handshake (stateless).
    func connect(serverURL: String, displayName: String) async throws -> ServerCapabilities {
        state = .connecting

        // 1. Get or create Ed25519 keypair (NET-04)
        let keypair = try await keyManager.getOrCreateKeypair()

        // 2. Connect WebSocket (NET-01)
        try await webSocket.connect(to: serverURL)
        state = .handshaking

        // 3. Build ainm.initialize request (NET-02)
        let identity = "\(displayName)@\(extractHost(from: serverURL))"
        let timestamp = ISO8601DateFormatter.ainm.string(from: Date())
        let initSignature = try MessageSigner.signInit(
            identity: identity,
            timestamp: timestamp,
            privateKey: keypair.privateKey
        )

        let initParams = InitializeParams(
            protocolVersion: "0.1.0",
            identity: identity,
            signature: initSignature,
            capabilities: ClientCapabilities(
                streaming: true,
                subThreads: false,
                contentTypes: ["text"],  // Phase 10: text only
                parallelStreaming: false,
                toolVisibility: false
            )
        )

        let requestId = nextRequestId
        nextRequestId += 1

        let request = JsonRpcRequest(
            method: "ainm.initialize",
            params: initParams,
            id: requestId
        )

        // 4. Send initialize request
        let requestData = try JSONEncoder().encode(request)
        try await webSocket.send(requestData)

        // 5. Wait for response
        let responseData = try await waitForResponse(id: requestId)
        let response = try JSONDecoder().decode(JsonRpcResponse.self, from: responseData)

        guard response.error == nil else {
            let errMsg = response.error?.message ?? "Unknown handshake error"
            state = .error(errMsg)
            throw ClientError.handshakeFailed(errMsg)
        }

        // 6. Decode server capabilities from result
        let resultData = try JSONEncoder().encode(response.result)
        let initResult = try JSONDecoder().decode(InitializeResult.self, from: resultData)

        // 7. Send ainm.initialized notification
        let notification = JsonRpcNotification.initialized()
        let notifData = try JSONEncoder().encode(notification)
        try await webSocket.send(notifData)

        // 8. Store capabilities and start receive loop
        serverCapabilities = initResult.capabilities
        state = .connected(initResult.capabilities)
        startReceiveLoop(publicKeys: [:])  // Phase 10: simplified key resolution

        await reconnection.reset()
        return initResult.capabilities
    }

    /// Send a signed message envelope (NET-05).
    func sendMessage(envelope: MessageEnvelope) async throws {
        guard case .connected = state else { throw ClientError.notConnected }

        let keypair = try await keyManager.getOrCreateKeypair()

        // Convert to dictionary for signing
        let envelopeData = try JSONEncoder().encode(envelope)
        guard var dict = try JSONSerialization.jsonObject(with: envelopeData) as? [String: Any] else {
            throw ClientError.encodingError
        }

        // Sign the envelope (NET-05)
        let signature = try MessageSigner.sign(envelope: dict, privateKey: keypair.privateKey)
        dict["signature"] = signature

        // Wrap in JSON-RPC request
        let requestId = nextRequestId
        nextRequestId += 1
        let rpcDict: [String: Any] = [
            "jsonrpc": "2.0",
            "id": requestId,
            "method": "ainm.message.send",
            "params": dict
        ]
        let rpcData = try JSONSerialization.data(withJSONObject: rpcDict)
        try await webSocket.send(rpcData)

        // Persist sent message to GRDB (write-through)
        var signedEnvelope = envelope
        signedEnvelope.signature = signature
        let record = try MessageRecord(from: signedEnvelope)
        try await appDatabase.dbPool.write { db in
            try record.save(db)
        }
    }

    /// Disconnect gracefully.
    func disconnect() async {
        receiveTask?.cancel()
        receiveTask = nil
        await webSocket.disconnect()
        state = .disconnected
    }

    // MARK: - Private

    private func waitForResponse(id: Int) async throws -> Data {
        for await result in await webSocket.receiveMessages() {
            switch result {
            case .success(let data):
                // Check if this is our response
                if let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any],
                   let responseId = json["id"] as? Int,
                   responseId == id {
                    return data
                }
                // Not our response -- skip (in production, queue for later)
            case .failure(let error):
                throw error
            }
        }
        throw ClientError.notConnected
    }

    private func startReceiveLoop(publicKeys: [String: Curve25519.Signing.PublicKey]) {
        receiveTask = Task { [weak self] in
            guard let self else { return }
            for await result in await webSocket.receiveMessages() {
                switch result {
                case .success(let data):
                    await self.handleIncomingMessage(data, publicKeys: publicKeys)
                case .failure:
                    // Disconnected -- trigger reconnection per D-06
                    await self.handleDisconnect()
                    break
                }
            }
        }
    }

    private func handleIncomingMessage(_ data: Data, publicKeys: [String: Curve25519.Signing.PublicKey]) async {
        guard let json = try? JSONSerialization.jsonObject(with: data) as? [String: Any] else { return }

        // Check if it's a notification (no "id" field = notification)
        if json["id"] == nil, let method = json["method"] as? String {
            switch method {
            case "ainm.message.deliver":
                if let params = json["params"] as? [String: Any] {
                    await handleMessageDelivery(params, publicKeys: publicKeys)
                }
            default:
                break  // Unknown notification, skip
            }
        }
    }

    private func handleMessageDelivery(_ params: [String: Any], publicKeys: [String: Curve25519.Signing.PublicKey]) async {
        // Write-through pattern: persist to GRDB first, then UI observes GRDB
        do {
            let paramsData = try JSONSerialization.data(withJSONObject: params)
            let envelope = try JSONDecoder().decode(MessageEnvelope.self, from: paramsData)
            let record = try MessageRecord(from: envelope)
            try await appDatabase.dbPool.write { db in
                try record.save(db)
            }
        } catch {
            // Log error but don't crash -- malformed messages are skipped
        }
    }

    private func handleDisconnect() async {
        state = .disconnected
        // D-06: auto-reconnect with exponential backoff
        // D-07: full re-handshake on reconnect
        // Reconnection needs the original URL and display name -- stored on first connect
        // For Phase 10, reconnection is best-effort. Full implementation in Phase 11.
    }

    private func extractHost(from urlString: String) -> String {
        URL(string: urlString)?.host ?? "localhost"
    }
}

// ISO8601 formatter matching protocol requirements
extension ISO8601DateFormatter {
    static let ainm: ISO8601DateFormatter = {
        let f = ISO8601DateFormatter()
        f.formatOptions = [.withInternetDateTime, .withFractionalSeconds]
        return f
    }()
}
```

```swift
import Foundation
import Observation

@Observable
final class AppState {
    var connectionState: ConnectionDisplayState = .disconnected
    var serverURL: String = "wss://localhost:8080"
    var displayName: String = ""
    var serverCapabilities: ServerCapabilities?
    var errorMessage: String?
    var connectedIdentity: String?

    enum ConnectionDisplayState: Sendable {
        case disconnected
        case connecting
        case connected
        case error
    }

    private var client: AINMClient?
    private var appDatabase: AppDatabase?

    func setup() throws {
        let db = try AppDatabase.makeDefault()
        self.appDatabase = db
        self.client = AINMClient(appDatabase: db)
    }

    func connect() async {
        guard let client else { return }
        connectionState = .connecting
        errorMessage = nil
        do {
            let capabilities = try await client.connect(
                serverURL: serverURL,
                displayName: displayName
            )
            serverCapabilities = capabilities
            connectedIdentity = "\(displayName)@\(URL(string: serverURL)?.host ?? "localhost")"
            connectionState = .connected
        } catch {
            connectionState = .error
            errorMessage = mapErrorMessage(error)
        }
    }

    func disconnect() async {
        await client?.disconnect()
        connectionState = .disconnected
        serverCapabilities = nil
        connectedIdentity = nil
    }

    func sendTestMessage() async throws {
        guard let client else { return }
        let envelope = MessageEnvelope(
            id: UUID().uuidString,  // Simplified for Phase 10; UUIDv7 in Phase 11
            from: connectedIdentity ?? "",
            threadId: "test-thread",
            timestamp: ISO8601DateFormatter.ainm.string(from: Date()),
            type: .message,
            content: [ContentBlock(type: "text", version: "1.0", source: "user", text: "Test message from iOS", fallback: "Test message from iOS")],
            signature: nil
        )
        try await client.sendMessage(envelope: envelope)
    }

    private func mapErrorMessage(_ error: Error) -> String {
        switch error {
        case WebSocketConnection.ConnectionError.invalidURL:
            return "Enter a valid WebSocket URL (wss:// or ws://)."
        case WebSocketConnection.ConnectionError.connectionFailed:
            return "Could not connect to server. Check the URL and try again."
        case AINMClient.ClientError.handshakeFailed(let msg):
            return "Server handshake failed. \(msg)"
        default:
            return "Could not connect to server. Check the URL and try again."
        }
    }
}
```
