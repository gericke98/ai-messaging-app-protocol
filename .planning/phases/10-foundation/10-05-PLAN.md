```swift
import SwiftUI

struct OnboardingView: View {
    @Bindable var appState: AppState

    var body: some View {
        VStack(spacing: 0) {
            Spacer()

            // App icon placeholder
            Image(systemName: "bubble.left.and.bubble.right.fill")
                .font(.system(size: 60))
                .foregroundStyle(Color.accentColor)
                .padding(.bottom, 16)  // md spacing

            // Title: .title2, bold, centered (per UI-SPEC)
            Text("AI-Native Messenger")
                .font(.title2)
                .bold()
                .padding(.bottom, 32)  // xl spacing

            // Form fields
            VStack(spacing: 16) {  // md spacing between fields
                VStack(alignment: .leading, spacing: 4) {  // xs spacing
                    Text("Server URL")
                        .font(.subheadline)
                    TextField("wss://localhost:8080", text: $appState.serverURL)
                        .keyboardType(.URL)
                        .textInputAutocapitalization(.never)
                        .autocorrectionDisabled()
                        .textFieldStyle(.roundedBorder)
                        .disabled(isConnecting)
                }

                VStack(alignment: .leading, spacing: 4) {
                    Text("Display Name")
                        .font(.subheadline)
                    TextField("Alice", text: $appState.displayName)
                        .textInputAutocapitalization(.words)
                        .textFieldStyle(.roundedBorder)
                        .disabled(isConnecting)
                }
            }
            .padding(.horizontal, 32)  // xl horizontal padding

            // 24pt gap before button (lg spacing per UI-SPEC)
            Spacer().frame(height: 24)

            // Connect button: .borderedProminent, 44pt min height
            Button {
                Task { await appState.connect() }
            } label: {
                HStack {
                    if isConnecting {
                        ProgressView()
                            .tint(.white)
                        Text("Connecting...")
                    } else {
                        Text("Connect")
                    }
                }
                .frame(maxWidth: .infinity)
                .frame(minHeight: 44)  // Apple HIG touch target
            }
            .buttonStyle(.borderedProminent)
            .disabled(!canConnect)
            .padding(.horizontal, 32)
            .accessibilityLabel("Connect to server")

            // Inline error message: .caption, Color.red (per UI-SPEC)
            if let error = appState.errorMessage {
                Text(error)
                    .font(.caption)
                    .foregroundStyle(Color.red)
                    .padding(.top, 8)
                    .padding(.horizontal, 32)
            }

            Spacer()
        }
        .background(Color(.systemBackground))
        .navigationDestination(isPresented: isConnectedBinding) {
            ConnectionSuccessView(appState: appState)
        }
    }

    private var isConnecting: Bool {
        appState.connectionState == .connecting
    }

    private var canConnect: Bool {
        !appState.serverURL.isEmpty &&
        !appState.displayName.isEmpty &&
        !isConnecting
    }

    private var isConnectedBinding: Binding<Bool> {
        Binding(
            get: { appState.connectionState == .connected },
            set: { if !$0 { /* handled by disconnect */ } }
        )
    }
}
```

**ConnectionSuccessView.swift** -- UI-SPEC Screen 2 (debug/validation):

```swift
import SwiftUI

struct ConnectionSuccessView: View {
    @Bindable var appState: AppState
    @State private var testResult: TestResult?

    enum TestResult {
        case success, failure(String)
    }

    var body: some View {
        VStack(spacing: 16) {
            // Title
            Text("Connected")
                .font(.title2)
                .bold()
                .padding(.top, 32)

            // Connection info
            VStack(alignment: .leading, spacing: 8) {
                infoRow("Server:", appState.serverURL)
                infoRow("Identity:", appState.connectedIdentity ?? "unknown")
                if let caps = appState.serverCapabilities {
                    infoRow("Streaming:", caps.streaming ? "supported" : "not supported")
                    infoRow("Max blocks:", "\(caps.maxContentBlocks)")
                    infoRow("Max message:", "\(caps.maxMessageSize) bytes")
                }
            }
            .padding(.horizontal, 32)

            Spacer().frame(height: 16)

            // Send Test Message button
            Button {
                Task { await sendTestMessage() }
            } label: {
                Text("Send Test Message")
                    .frame(maxWidth: .infinity)
                    .frame(minHeight: 44)
            }
            .buttonStyle(.borderedProminent)
            .padding(.horizontal, 32)

            // Test result
            if let result = testResult {
                switch result {
                case .success:
                    Label("Message signed and verified", systemImage: "checkmark.circle.fill")
                        .font(.caption)
                        .foregroundStyle(Color.green)
                case .failure(let msg):
                    Label("Signature verification failed: \(msg)", systemImage: "xmark.circle.fill")
                        .font(.caption)
                        .foregroundStyle(Color.red)
                }
            }

            Spacer()

            // Disconnect button: .bordered, red tint
            Button {
                Task { await appState.disconnect() }
            } label: {
                Text("Disconnect")
                    .frame(maxWidth: .infinity)
                    .frame(minHeight: 44)
            }
            .buttonStyle(.bordered)
            .tint(Color.red)
            .padding(.horizontal, 32)
            .padding(.bottom, 32)
        }
        .background(Color(.systemBackground))
        .navigationBarBackButtonHidden(true)
    }

    private func infoRow(_ label: String, _ value: String) -> some View {
        HStack {
            Text(label)
                .font(.body)
                .foregroundStyle(Color.secondary)
            Text(value)
                .font(.body)
        }
    }

    private func sendTestMessage() async {
        do {
            try await appState.sendTestMessage()
            testResult = .success
        } catch {
            testResult = .failure(error.localizedDescription)
        }
    }
}
```

Create GitHub Actions CI workflow per D-10: builds iOS project and runs unit tests on every push.

**.github/workflows/ios-ci.yml:**
```yaml
name: iOS CI

on:
  push:
    branches: [main]
    paths:
      - 'ios/**'
      - 'tests/vectors/**'
      - '.github/workflows/ios-ci.yml'
  pull_request:
    branches: [main]
    paths:
      - 'ios/**'
      - 'tests/vectors/**'

jobs:
  build-and-test:
    runs-on: macos-14
    timeout-minutes: 30

    steps:
      - name: Checkout
        uses: actions/checkout@v4

      - name: Select Xcode 16
        run: sudo xcode-select -s /Applications/Xcode_16.0.app/Contents/Developer

      - name: Show Xcode version
        run: xcodebuild -version

      - name: Resolve SPM dependencies
        working-directory: ios/AINMMessenger
        run: xcodebuild -resolvePackageDependencies -project AINMMessenger.xcodeproj -scheme AINMMessenger

      - name: Build
        working-directory: ios/AINMMessenger
        run: |
          xcodebuild build \
            -project AINMMessenger.xcodeproj \
            -scheme AINMMessenger \
            -destination 'platform=iOS Simulator,name=iPhone 16,OS=18.0' \
            CODE_SIGNING_ALLOWED=NO

      - name: Test
        working-directory: ios/AINMMessenger
        run: |
          xcodebuild test \
            -project AINMMessenger.xcodeproj \
            -scheme AINMMessenger \
            -destination 'platform=iOS Simulator,name=iPhone 16,OS=18.0' \
            CODE_SIGNING_ALLOWED=NO \
            -resultBundlePath TestResults
        continue-on-error: false
```

Key decisions per D-10:
- `macos-14` runner (Apple Silicon, has Xcode 16)
- Triggers on push to main and PRs, scoped to ios/ and tests/vectors/ paths
- `CODE_SIGNING_ALLOWED=NO` avoids needing provisioning profiles in CI
- 30-minute timeout (SPM resolution + build + test)
- Xcode 16 selected explicitly
