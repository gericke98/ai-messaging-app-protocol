**KeyManager.swift:**
```swift
import CryptoKit
import Foundation

actor KeyManager {
    private static let keychainService = "com.ainm.messenger"
    private static let keychainAccount = "ed25519.privateKey"

    struct KeyPairInfo: Sendable {
        let privateKey: Curve25519.Signing.PrivateKey
        let publicKey: Curve25519.Signing.PublicKey
        let publicKeyBase64url: String
    }

    /// Get existing keypair from Keychain, or generate a new one and store it.
    func getOrCreateKeypair() throws -> KeyPairInfo {
        if let existingData = try loadPrivateKeyData() {
            let privateKey = try Curve25519.Signing.PrivateKey(rawRepresentation: existingData)
            return makeInfo(privateKey)
        }
        let privateKey = Curve25519.Signing.PrivateKey()
        try savePrivateKeyData(privateKey.rawRepresentation)
        return makeInfo(privateKey)
    }

    /// Delete stored keypair (for testing or key rotation).
    func deleteKeypair() throws { ... }

    private func makeInfo(_ pk: Curve25519.Signing.PrivateKey) -> KeyPairInfo {
        KeyPairInfo(
            privateKey: pk,
            publicKey: pk.publicKey,
            publicKeyBase64url: pk.publicKey.rawRepresentation.base64urlEncodedString()
        )
    }
}
```

**MessageSigner.swift** -- follows security.md Section 4 (signing) and Section 5 (verification) exactly, matching `demo/src/signing.rs`:

```swift
import CryptoKit
import Foundation

struct MessageSigner: Sendable {

    enum SigningError: Error {
        case encodingFailed
        case missingSignature
        case invalidSignature
        case canonicalizationFailed
    }

    /// Sign a message envelope per security.md Section 4.
    /// Input: Dictionary representation of envelope WITHOUT "signature" key.
    /// Output: base64url-encoded Ed25519 signature.
    ///
    /// Steps (matching signing.rs sign_envelope):
    /// 1. Remove "signature" from input (defensive)
    /// 2. Canonicalize with RFC 8785 JCS
    /// 3. Encode canonical string as UTF-8 bytes
    /// 4. Sign bytes with Ed25519 private key
    /// 5. Return base64url-encoded signature (no padding)
    static func sign(
        envelope: [String: Any],
        privateKey: Curve25519.Signing.PrivateKey
    ) throws -> String {
        var signingInput = envelope
        signingInput.removeValue(forKey: "signature")
        let canonical = try JCSCanonicalizer.canonicalize(signingInput)
        guard let data = canonical.data(using: .utf8) else {
            throw SigningError.encodingFailed
        }
        let signature = try privateKey.signature(for: data)
        return signature.withUnsafeBytes { Data($0) }.base64urlEncodedString()
    }

    /// Verify a message envelope's signature per security.md Section 5.
    /// Input: Dictionary with "signature" key present.
    /// Output: true if signature is valid.
    static func verify(
        envelope: [String: Any],
        publicKey: Curve25519.Signing.PublicKey
    ) throws -> Bool {
        guard let sigString = envelope["signature"] as? String else {
            throw SigningError.missingSignature
        }
        var signingInput = envelope
        signingInput.removeValue(forKey: "signature")
        let canonical = try JCSCanonicalizer.canonicalize(signingInput)
        guard let data = canonical.data(using: .utf8) else {
            throw SigningError.encodingFailed
        }
        guard let sigData = Data(base64urlEncoded: sigString) else {
            throw SigningError.invalidSignature
        }
        return publicKey.isValidSignature(sigData, for: data)
    }

    /// Sign an initialization request per transport.md Section 4.3.
    /// The init token is: identity_address + "|" + timestamp
    static func signInit(
        identity: String,
        timestamp: String,
        privateKey: Curve25519.Signing.PrivateKey
    ) throws -> String {
        let payload = "\(identity)|\(timestamp)"
        guard let data = payload.data(using: .utf8) else {
            throw SigningError.encodingFailed
        }
        let signature = try privateKey.signature(for: data)
        return signature.withUnsafeBytes { Data($0) }.base64urlEncodedString()
    }
}
```
