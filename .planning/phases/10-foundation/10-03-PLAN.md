Create `AppDatabase.swift` that sets up a GRDB `DatabasePool` with versioned migrations.

```swift
import GRDB
import Foundation

struct AppDatabase: Sendable {
    let dbPool: DatabasePool

    /// Create database at the standard Application Support path.
    /// Creates the directory if it doesn't exist.
    static func makeDefault() throws -> AppDatabase {
        let fileManager = FileManager.default
        let appSupportURL = try fileManager.url(
            for: .applicationSupportDirectory,
            in: .userDomainMask,
            appropriateFor: nil,
            create: true
        )
        let dbURL = appSupportURL.appendingPathComponent("ainm.sqlite")
        let dbPool = try DatabasePool(path: dbURL.path)
        let db = AppDatabase(dbPool: dbPool)
        try db.migrate()
        return db
    }

    /// Create in-memory database for testing.
    static func makeInMemory() throws -> AppDatabase {
        let dbPool = try DatabasePool(path: ":memory:")
        let db = AppDatabase(dbPool: dbPool)
        try db.migrate()
        return db
    }

    private func migrate() throws {
        var migrator = DatabaseMigrator()

        // Always run all migrations in development (wipe on schema change).
        // Remove this line for production releases.
        #if DEBUG
        migrator.eraseDatabaseOnSchemaChange = true
        #endif

        migrator.registerMigration("v1_create_threads") { db in
            try db.create(table: "thread") { t in
                t.primaryKey("id", .text).notNull()              // UUIDv7 string
                t.column("title", .text)                         // Optional thread title
                t.column("createdAt", .text).notNull()           // RFC 3339 timestamp
                t.column("createdBy", .text).notNull()           // AINM address
                t.column("parentThreadId", .text)                // Sub-thread parent
                    .references("thread", onDelete: .cascade)
                t.column("parentMessageId", .text)               // Message that spawned sub-thread
            }
        }

        migrator.registerMigration("v1_create_messages") { db in
            try db.create(table: "message") { t in
                t.primaryKey("id", .text).notNull()              // UUIDv7 string
                t.column("threadId", .text).notNull()            // FK to thread
                    .references("thread", onDelete: .cascade)
                t.column("fromAddress", .text).notNull()         // AINM address of sender
                t.column("timestamp", .text).notNull()           // RFC 3339 timestamp
                t.column("type", .text).notNull()                // "message", "thread.create", etc.
                t.column("contentJson", .text).notNull()         // JSON array of content blocks
                t.column("signature", .text).notNull()           // base64url Ed25519 signature
                t.column("serverTimestamp", .text)               // Server-assigned timestamp
            }
            // Index for querying messages by thread, ordered by timestamp
            try db.create(index: "idx_message_thread_timestamp",
                          on: "message",
                          columns: ["threadId", "timestamp"])
        }

        migrator.registerMigration("v1_create_participants") { db in
            try db.create(table: "participant") { t in
                t.column("threadId", .text).notNull()            // FK to thread
                    .references("thread", onDelete: .cascade)
                t.column("identity", .text).notNull()            // AINM address
                t.column("type", .text).notNull()                // "human" or "agent"
                t.column("joinedAt", .text).notNull()            // RFC 3339 timestamp
                t.column("role", .text).notNull()                // "creator" or "member"
                t.primaryKey(["threadId", "identity"])
            }
        }

        try migrator.migrate(dbPool)
    }
}
```

Key design decisions:
- All timestamp columns are `.text` (RFC 3339 strings), not `.datetime`, because the protocol uses ISO 8601 with milliseconds and we want lossless round-tripping.
- `contentJson` stores the full content blocks array as a JSON string -- avoids a separate content_block table for Phase 10 simplicity. Phase 11 can add a separate table if querying individual blocks is needed.
- Foreign keys with `onDelete: .cascade` ensure orphaned messages/participants are cleaned up.
- Composite primary key on participant (threadId + identity) prevents duplicate participants.
- Index on (threadId, timestamp) for efficient message list queries.

**MessageRecord.swift:**
```swift
import GRDB
import Foundation

struct MessageRecord: Codable, Sendable, FetchableRecord, PersistableRecord {
    static let databaseTableName = "message"
    static let threadForeignKey = ForeignKey(["threadId"], to: ["id"])

    let id: String
    let threadId: String
    let fromAddress: String
    let timestamp: String
    let type: String
    let contentJson: String     // JSON-encoded [ContentBlock]
    let signature: String
    var serverTimestamp: String?

    /// Create from protocol MessageEnvelope
    init(from envelope: MessageEnvelope) throws {
        self.id = envelope.id
        self.threadId = envelope.threadId
        self.fromAddress = envelope.from
        self.timestamp = envelope.timestamp
        self.type = envelope.type.rawValue
        self.contentJson = try String(
            data: JSONEncoder().encode(envelope.content),
            encoding: .utf8
        )!
        self.signature = envelope.signature ?? ""
        self.serverTimestamp = nil
    }
}

// Query helpers
extension MessageRecord {
    /// Fetch messages for a thread, ordered by timestamp ascending.
    static func messagesForThread(_ threadId: String) -> QueryInterfaceRequest<MessageRecord> {
        filter(Column("threadId") == threadId)
            .order(Column("timestamp").asc)
    }
}
```
