    Create website/scripts/copy-specs.js:
    ```javascript
    const fs = require('fs');
    const path = require('path');

    // Copy spec/ files to docs/spec/ with Docusaurus frontmatter
    const specDir = path.join(__dirname, '../../spec');
    const docsSpecDir = path.join(__dirname, '../docs/spec');
    fs.mkdirSync(docsSpecDir, { recursive: true });

    const specOrder = [
      'identity', 'security', 'transport', 'content', 'interactive',
      'threading', 'streaming', 'agent-card', 'memory', 'tool-visibility', 'compatibility'
    ];

    specOrder.forEach((name, index) => {
      const src = path.join(specDir, `${name}.md`);
      const dest = path.join(docsSpecDir, `${name}.md`);
      let content = fs.readFileSync(src, 'utf8');

      // Strip existing non-Docusaurus headers (Status, Version, Last Updated lines)
      // and prepend Docusaurus frontmatter
      const title = content.match(/^#\s+(.+)/m)?.[1] || name;
      const frontmatter = `---
sidebar_position: ${index + 1}
title: "${title}"
---

`;
      // Remove the first H1 line (Docusaurus uses title from frontmatter)
      content = content.replace(/^#\s+.+\n/, '');

      fs.writeFileSync(dest, frontmatter + content);
    });

    // Copy schemas to static/schemas for download
    const schemasDir = path.join(__dirname, '../../schemas');
    const staticSchemasDir = path.join(__dirname, '../static/schemas');
    fs.mkdirSync(staticSchemasDir, { recursive: true });

    if (fs.existsSync(schemasDir)) {
      fs.readdirSync(schemasDir).forEach(file => {
        if (file.endsWith('.json')) {
          fs.copyFileSync(
            path.join(schemasDir, file),
            path.join(staticSchemasDir, file)
          );
        }
      });
    }

    console.log('Copied spec files and schemas');
    ```

    **Create .github/workflows/deploy-docs.yml:**

    ```yaml
    name: Deploy Spec Site to GitHub Pages

    on:
      push:
        branches: [main]
        paths:
          - 'spec/**'
          - 'schemas/**'
          - 'asyncapi.yaml'
          - 'website/**'
          - 'CHANGELOG.md'
      workflow_dispatch:

    permissions:
      contents: read
      pages: write
      id-token: write

    concurrency:
      group: "pages"
      cancel-in-progress: false

    jobs:
      build:
        runs-on: ubuntu-latest
        steps:
          - name: Checkout
            uses: actions/checkout@v4

          - name: Setup Node.js
            uses: actions/setup-node@v4
            with:
              node-version: 20
              cache: npm
              cache-dependency-path: website/package-lock.json

          - name: Install dependencies
            working-directory: website
            run: npm ci

          - name: Validate AsyncAPI
            working-directory: website
            run: npx asyncapi validate ../asyncapi.yaml

          - name: Build site
            working-directory: website
            run: npm run build
            env:
              NODE_OPTIONS: --max-old-space-size=4096

          - name: Upload artifact
            uses: actions/upload-pages-artifact@v3
            with:
              path: website/build

      deploy:
        environment:
          name: github-pages
          url: ${{ steps.deployment.outputs.page_url }}
        runs-on: ubuntu-latest
        needs: build
        steps:
          - name: Deploy to GitHub Pages
            id: deployment
            uses: actions/deploy-pages@v4
    ```

    Key design choices:
    - Triggers on push to main when spec/, schemas/, asyncapi.yaml, website/, or CHANGELOG.md change
    - Also has workflow_dispatch for manual triggers
    - Validates asyncapi.yaml with @asyncapi/cli before build
    - Uses Node.js 20 (LTS)
    - Builds with increased memory for AsyncAPI component processing
    - Uses official GitHub Pages actions (upload-pages-artifact + deploy-pages)
    - Concurrency group prevents multiple deploys

    **After creating the workflow, verify the site builds locally:**
    Run `cd website && npm run build` to confirm the Docusaurus site builds without errors.
    If the build fails due to the asyncapi preset, investigate and fix. Per Pitfall 1, the issue would be webpack Node.js polyfill errors -- the fix is ensuring docusaurus-asyncapi@1.0.2 is installed correctly.
