```rust
use std::collections::{HashMap, HashSet};
use std::sync::{Arc, Mutex};

#[derive(Clone, Default)]
pub struct ThreadRegistry {
    inner: Arc<Mutex<HashMap<String, ThreadInfo>>>,
}

pub struct ThreadInfo {
    pub origin_domain: String,
    pub participating_servers: HashSet<String>,
    pub sequence_counter: u64,
}

impl ThreadRegistry {
    pub fn new() -> Self { Self { inner: Arc::new(Mutex::new(HashMap::new())) } }

    /// Create a new thread owned by this server. Per D-10.
    pub fn create_thread(&self, origin_domain: &str) -> String {
        let thread_id = format!("{}:{}", origin_domain, uuid::Uuid::now_v7());
        let info = ThreadInfo {
            origin_domain: origin_domain.to_string(),
            participating_servers: HashSet::new(),
            sequence_counter: 0,
        };
        self.inner.lock().unwrap().insert(thread_id.clone(), info);
        thread_id
    }

    /// Add a remote server to a thread's participant set.
    pub fn add_server(&self, thread_id: &str, server_id: &str) {
        if let Some(info) = self.inner.lock().unwrap().get_mut(thread_id) {
            info.participating_servers.insert(server_id.to_string());
        }
    }

    /// Get next sequence number (origin only). Per D-11.
    pub fn next_sequence(&self, thread_id: &str) -> Option<u64> {
        let mut map = self.inner.lock().unwrap();
        map.get_mut(thread_id).map(|info| {
            info.sequence_counter += 1;
            info.sequence_counter
        })
    }

    /// Get thread info (origin domain, participating servers).
    pub fn get_info(&self, thread_id: &str) -> Option<(String, HashSet<String>)> {
        self.inner.lock().unwrap().get(thread_id).map(|info| {
            (info.origin_domain.clone(), info.participating_servers.clone())
        })
    }

    /// Register a thread we learned about from a relay (not origin).
    pub fn register_remote_thread(&self, thread_id: &str, origin_domain: &str) {
        let mut map = self.inner.lock().unwrap();
        if !map.contains_key(thread_id) {
            map.insert(thread_id.to_string(), ThreadInfo {
                origin_domain: origin_domain.to_string(),
                participating_servers: HashSet::new(),
                sequence_counter: 0,
            });
        }
    }
}
```

```rust
use axum::{extract::State, http::StatusCode, Json};

pub async fn relay_handler(
    State(state): State<FederationState>,
    Json(body): Json<serde_json::Value>,
) -> (StatusCode, Json<serde_json::Value>) {
    // 1. Parse JSON-RPC envelope
    let method = body.get("method").and_then(|m| m.as_str()).unwrap_or("");
    let id = body.get("id").cloned().unwrap_or(serde_json::json!(1));

    if method != "ainm.relay" {
        return (StatusCode::OK, Json(protocol::new_error_value(id, -32601, "Method not found")));
    }

    let params = match body.get("params") {
        Some(p) => p.clone(),
        None => return (StatusCode::OK, Json(protocol::new_error_value(id, -32602, "Missing params"))),
    };

    let from_server = match params.get("from_server").and_then(|s| s.as_str()) {
        Some(s) => s.to_string(),
        None => return (StatusCode::OK, Json(protocol::new_error_value(id, -32602, "Missing from_server"))),
    };

    // 2. Verify server signature (FAUT-03 layer 2)
    let remote_identity = match state.remote_cache.get(&from_server) {
        Some(id) => id,
        None => {
            // Try to resolve
            tracing::warn!("Unknown server: {}", from_server);
            return (StatusCode::OK, Json(protocol::new_error_value(id, -32011, "Unknown server identity")));
        }
    };

    let remote_pk_bytes = base64::engine::general_purpose::URL_SAFE_NO_PAD
        .decode(&remote_identity.public_key_b64)
        .unwrap_or_default();

    if !crate::signing::verify_relay_signature(&remote_pk_bytes, &params).unwrap_or(false) {
        tracing::error!("Server signature verification failed for {}", from_server);
        return (StatusCode::OK, Json(protocol::new_error_value(id, -32010, "Server signature verification failed")));
    }
    tracing::info!("[federation] Server signature verified: {}", from_server);

    // 3. Verify individual message signatures (FAUT-03 layer 3)
    let messages = params.get("messages").and_then(|m| m.as_array()).cloned().unwrap_or_default();
    for msg in &messages {
        let sender = msg.get("from").and_then(|f| f.as_str()).unwrap_or("");
        if let Some(pk_bytes) = state.identity_store.get_public_key_bytes(sender) {
            if !crate::signing::verify_signature(&pk_bytes, msg).unwrap_or(false) {
                tracing::error!("Message signature failed for sender: {}", sender);
                return (StatusCode::OK, Json(protocol::new_error_value(id, -32010, "Message signature verification failed")));
            }
            tracing::info!("[federation] Message signature verified: {}", sender);
        } else {
            tracing::warn!("Unknown sender identity: {}", sender);
            // For Phase 7 demo, reject unknown senders
            return (StatusCode::OK, Json(protocol::new_error_value(id, -32011, "Unknown sender identity")));
        }
    }

    // 4. Process messages: register thread, deliver locally, fan out if origin
    for msg in &messages {
        let thread_id = msg.get("thread_id").and_then(|t| t.as_str()).unwrap_or("");
        let origin = parse_origin_domain(thread_id).unwrap_or("");

        // Register the thread if we don't know about it
        state.thread_registry.register_remote_thread(thread_id, origin);
        state.thread_registry.add_server(thread_id, &from_server);

        // Deliver to local participants via broadcast channel
        let deliver = serde_json::json!({
            "jsonrpc": "2.0",
            "method": "ainm.message.deliver",
            "params": {
                "message": msg,
                "verified": true,
                "relayed_from": from_server,
            }
        });
        let _ = state.local_delivery_tx.send(serde_json::to_string(&deliver).unwrap());

        // If we are the origin server, assign sequence and fan out to other servers
        if origin == state.server_id {
            if let Some(seq) = state.thread_registry.next_sequence(thread_id) {
                tracing::info!("[federation] Origin assigned seq {} for thread {}", seq, thread_id);
                // Fan out to other participating servers (skip sender)
                // For Phase 7 2-server demo, there's only one remote, but the structure supports N
                if let Some(sender) = &state.relay_sender {
                    if let Some((_, servers)) = state.thread_registry.get_info(thread_id) {
                        for server in servers {
                            if server != from_server {
                                sender.enqueue(&server, msg.clone());
                            }
                        }
                    }
                }
            }
        }
    }

    // 5. Return success
    let result = serde_json::json!({
        "jsonrpc": "2.0",
        "id": id,
        "result": {
            "accepted": true,
            "messages_processed": messages.len()
        }
    });
    (StatusCode::OK, Json(result))
}
```

```rust
use std::time::{Duration, Instant};

struct BackoffState {
    attempt: u32,
    base_delay: Duration,
    max_delay: Duration,
    next_retry: Option<Instant>,
}

impl BackoffState {
    fn new() -> Self {
        Self { attempt: 0, base_delay: Duration::from_secs(1), max_delay: Duration::from_secs(60), next_retry: None }
    }
    fn record_failure(&mut self) {
        self.attempt += 1;
        let delay = self.base_delay * 2u32.pow(self.attempt.min(6));
        self.next_retry = Some(Instant::now() + delay.min(self.max_delay));
    }
    fn record_success(&mut self) { self.attempt = 0; self.next_retry = None; }
    fn can_retry(&self) -> bool { self.next_retry.map_or(true, |t| Instant::now() >= t) }
}
```

3. **RelaySender** struct (the main public interface):
```rust
use std::collections::HashMap;
use std::sync::{Arc, Mutex};

pub struct RelaySender {
    queues: Arc<Mutex<HashMap<String, DestinationQueue>>>,
    http_client: reqwest::Client,
    server_keypair: Arc<crate::identity::Identity>,
    server_id: String,
    remote_cache: crate::identity::RemoteServerCache,
}

impl RelaySender {
    pub fn new(
        http_client: reqwest::Client,
        server_keypair: Arc<crate::identity::Identity>,
        server_id: String,
        remote_cache: crate::identity::RemoteServerCache,
    ) -> Self {
        Self {
            queues: Arc::new(Mutex::new(HashMap::new())),
            http_client,
            server_keypair,
            server_id,
            remote_cache,
        }
    }

    /// Enqueue a message for delivery to a remote server. Per RELY-04: push immediately.
    pub fn enqueue(&self, destination: &str, message: serde_json::Value) {
        let mut queues = self.queues.lock().unwrap();
        let queue = queues.entry(destination.to_string()).or_insert_with(|| {
            DestinationQueue { messages: VecDeque::new(), backoff: BackoffState::new() }
        });
        queue.messages.push_back(message);
        tracing::info!("[relay] Enqueued message for {}", destination);
    }

    /// Send all queued messages for a destination. Called by the send loop.
    pub async fn flush_destination(&self, destination: &str) -> Result<(), String> {
        let messages: Vec<serde_json::Value> = {
            let mut queues = self.queues.lock().unwrap();
            if let Some(queue) = queues.get_mut(destination) {
                if !queue.backoff.can_retry() {
                    return Ok(()); // Still in backoff
                }
                queue.messages.drain(..).collect()
            } else {
                return Ok(());
            }
        };

        if messages.is_empty() { return Ok(()); }

        // Look up remote server's relay endpoint
        let remote = self.remote_cache.get(destination)
            .ok_or_else(|| format!("Unknown destination: {}", destination))?;

        // Build relay payload
        let mut params = serde_json::json!({
            "messages": messages,
            "from_server": self.server_id,
        });

        // Sign relay params with server key
        crate::signing::sign_relay_params(&self.server_keypair.key_pair, &mut params)
            .map_err(|e| format!("Signing failed: {}", e))?;

        let payload = serde_json::json!({
            "jsonrpc": "2.0",
            "id": 1,
            "method": "ainm.relay",
            "params": params,
        });

        tracing::info!("[relay] Sending {} message(s) to {}", messages.len(), remote.relay_endpoint);

        // POST to relay endpoint
        match self.http_client.post(&remote.relay_endpoint)
            .json(&payload)
            .send()
            .await
        {
            Ok(resp) if resp.status().is_success() => {
                let mut queues = self.queues.lock().unwrap();
                if let Some(queue) = queues.get_mut(destination) {
                    queue.backoff.record_success();
                }
                tracing::info!("[relay] Successfully relayed to {}", destination);
                Ok(())
            }
            Ok(resp) => {
                let status = resp.status();
                let body = resp.text().await.unwrap_or_default();
                tracing::error!("[relay] Relay to {} failed: {} - {}", destination, status, body);
                // Re-enqueue messages and record failure
                let mut queues = self.queues.lock().unwrap();
                if let Some(queue) = queues.get_mut(destination) {
                    for msg in messages.into_iter().rev() {
                        queue.messages.push_front(msg);
                    }
                    queue.backoff.record_failure();
                }
                Err(format!("HTTP {}", status))
            }
            Err(e) => {
                tracing::error!("[relay] Connection to {} failed: {}", destination, e);
                let mut queues = self.queues.lock().unwrap();
                if let Some(queue) = queues.get_mut(destination) {
                    for msg in messages.into_iter().rev() {
                        queue.messages.push_front(msg);
                    }
                    queue.backoff.record_failure();
                }
                Err(format!("Connection error: {}", e))
            }
        }
    }

    /// Get list of destinations with pending messages.
    pub fn pending_destinations(&self) -> Vec<String> {
        self.queues.lock().unwrap()
            .iter()
            .filter(|(_, q)| !q.messages.is_empty())
            .map(|(k, _)| k.clone())
            .collect()
    }

    /// Start the background send loop. Checks all queues every 500ms.
    pub fn start_send_loop(self: &Arc<Self>) -> tokio::task::JoinHandle<()> {
        let sender = Arc::clone(self);
        tokio::spawn(async move {
            loop {
                let destinations = sender.pending_destinations();
                for dest in destinations {
                    let _ = sender.flush_destination(&dest).await;
                }
                tokio::time::sleep(Duration::from_millis(500)).await;
            }
        })
    }
}
```
