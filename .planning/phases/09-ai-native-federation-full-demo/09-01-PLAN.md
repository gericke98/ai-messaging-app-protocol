```rust
let agent_card_for_fed = agent_card.clone();
let fed_agent_card_handler = move |headers: axum::http::HeaderMap| {
    let card = agent_card_for_fed.clone();
    async move {
        let body = serde_json::to_vec(&card).unwrap();
        let hash = ring::digest::digest(&ring::digest::SHA256, &body);
        let etag_value = format!("\"{}\"", base64::engine::general_purpose::URL_SAFE_NO_PAD.encode(hash.as_ref()));
        if let Some(if_none_match) = headers.get(axum::http::header::IF_NONE_MATCH) {
            if if_none_match.to_str().unwrap_or("") == etag_value {
                return axum::http::StatusCode::NOT_MODIFIED.into_response();
            }
        }
        ([(axum::http::header::CONTENT_TYPE, "application/json"), (axum::http::header::ETAG, &etag_value)], body).into_response()
    }
};
```

Then add to `fed_app`:

**server.rs — Relay streaming and tool events when thread is federated (per D-04/D-05/D-06):**

In `handle_agent_connection`, replace the streaming passthrough block (lines 618-623) with federation-aware relay logic:

```rust
"ainm.stream.start" | "ainm.stream.delta" | "ainm.stream.end"
| "ainm.presence.update" => {
    // Always deliver locally
    let _ = to_human_tx.send(text_str.clone());

    // Federation relay if thread is federated (D-04: stream events via ainm.relay)
    if let Some(ref fed) = state.federation {
        // StreamStart has thread_id in params; delta/end only have stream_id
        // For start: extract thread_id directly
        // For delta/end: use tracked stream->thread mapping
        let thread_id_opt = parsed.get("params")
            .and_then(|p| p.get("thread_id"))
            .and_then(|t| t.as_str());

        let effective_thread_id = if let Some(tid) = thread_id_opt {
            // StreamStart — track the stream_id -> thread_id mapping
            if method == "ainm.stream.start" {
                if let Some(sid) = parsed.get("params").and_then(|p| p.get("stream_id")).and_then(|s| s.as_str()) {
                    stream_thread_map.insert(sid.to_string(), tid.to_string());
                }
            }
            Some(tid.to_string())
        } else {
            // Delta/End — look up thread_id from stream_id
            parsed.get("params")
                .and_then(|p| p.get("stream_id"))
                .and_then(|s| s.as_str())
                .and_then(|sid| stream_thread_map.get(sid).cloned())
        };

        if let Some(tid) = effective_thread_id {
            let origin = federation::parse_origin_domain(&tid).unwrap_or("");
            if !origin.is_empty() && origin != fed.server_id {
                // Relay to origin server (which will fan out)
                fed.relay_sender.enqueue(origin, parsed.clone());
            } else if origin == fed.server_id {
                // We ARE origin — fan out to all participating remote servers
                if let Some((_, remotes)) = fed.thread_registry.get_info(&tid) {
                    for remote in &remotes {
                        if remote != &fed.server_id {
                            fed.relay_sender.enqueue(remote, parsed.clone());
                        }
                    }
                }
            }
        }

        // Clean up stream_thread_map on StreamEnd
        if method == "ainm.stream.end" {
            if let Some(sid) = parsed.get("params").and_then(|p| p.get("stream_id")).and_then(|s| s.as_str()) {
                stream_thread_map.remove(sid);
            }
        }
    }
}
```

```rust
line if line.starts_with("/discover ") => {
    let target = line.trim_start_matches("/discover ").trim();
    if target.is_empty() {
        println!("  Usage: /discover agent@server-domain");
        continue;
    }
    // Parse agent@domain format
    let parts: Vec<&str> = target.splitn(2, '@').collect();
    if parts.len() != 2 {
        println!("  Invalid format. Use: /discover agent@server-domain");
        continue;
    }
    let agent_name = parts[0];
    let domain = parts[1];

    let url = format!(
        "https://{}/.well-known/ainm/{}/agent-card.json",
        domain, agent_name
    );
    println!("  Discovering {}...", target);
    let rt = tokio::runtime::Handle::current();
    match rt.block_on(fetch_agent_card(&url)) {
        Ok(card) => {
            println!("  Agent Card resolved for {}:", target);
            render::render_agent_card(&card);
        }
        Err(e) => println!("  Failed to discover {}: {}", target, e),
    }
}
```

```rust
async fn fetch_remote_agent_card(url: &str) -> Result<serde_json::Value> {
```
