4. Update `enqueue()` to use persistence and enforce 10K bound (D-11, D-13):
   ```rust
   pub fn enqueue(&self, destination: &str, message: serde_json::Value) {
       // Persist to redb (FREL-01: survives restart)
       if let Err(e) = self.persistence.enqueue_offline(destination, &message) {
           tracing::error!("[relay] Failed to persist message for {}: {}", destination, e);
           return;
       }

       // Enforce 10K bound per destination (D-11: FIFO eviction)
       match self.persistence.queue_len(destination) {
           Ok(len) if len > 10_000 => {
               match self.persistence.evict_oldest(destination, 10_000) {
                   Ok(evicted) => {
                       tracing::warn!(
                           "[relay] Queue bound exceeded for {}: evicted {} oldest messages",
                           destination, evicted
                       );
                   }
                   Err(e) => tracing::error!("[relay] Eviction failed for {}: {}", destination, e),
               }
           }
           _ => {}
       }

       tracing::info!("[relay] Enqueued message for {} (persistent)", destination);
   }
   ```

5. Update `flush_destination()` to drain from persistence:
   ```rust
   pub async fn flush_destination(&self, destination: &str) -> Result<(), String> {
       // Check backoff
       {
           let backoffs = self.backoff_states.lock().unwrap();
           if let Some(backoff) = backoffs.get(destination) {
               if !backoff.can_retry() {
                   return Ok(());
               }
           }
       }

       // Drain messages from persistent queue (batch of up to 100)
       let messages = self.persistence.drain_offline(destination, 100)?;
       if messages.is_empty() {
           return Ok(());
       }

       // Look up remote server's relay endpoint
       let remote = self.remote_cache.get(destination)
           .ok_or_else(|| format!("Unknown destination: {}", destination))?;

       // Build relay payload per transport.md Section 5.3
       let mut params = serde_json::json!({
           "messages": messages,
           "from_server": self.server_id,
           "relay_timestamp": chrono::Utc::now().to_rfc3339_opts(chrono::SecondsFormat::Millis, true),
           "relay_id": uuid::Uuid::now_v7().to_string(),
       });

       // Sign relay params
       crate::signing::sign_relay_params(&self.server_keypair.key_pair, &mut params)
           .map_err(|e| format!("Signing failed: {}", e))?;

       let payload = serde_json::json!({
           "jsonrpc": "2.0",
           "id": 1,
           "method": "ainm.relay",
           "params": params,
       });

       let msg_count = messages.len();
       tracing::info!("[relay] Sending {} message(s) to {}", msg_count, remote.relay_endpoint);

       match self.http_client.post(&remote.relay_endpoint).json(&payload).send().await {
           Ok(resp) if resp.status().is_success() => {
               // Success: reset backoff (D-14: reset on reconnection)
               let mut backoffs = self.backoff_states.lock().unwrap();
               backoffs.entry(destination.to_string())
                   .or_insert_with(BackoffState::new)
                   .record_success();
               tracing::info!("[relay] Successfully relayed to {}", destination);
               Ok(())
           }
           Ok(resp) => {
               let status = resp.status();
               let body = resp.text().await.unwrap_or_default();
               tracing::error!("[relay] Relay to {} failed: {} - {}", destination, status, body);
               // Re-enqueue messages back to persistence
               for msg in messages {
                   let _ = self.persistence.enqueue_offline(destination, &msg);
               }
               let mut backoffs = self.backoff_states.lock().unwrap();
               backoffs.entry(destination.to_string())
                   .or_insert_with(BackoffState::new)
                   .record_failure();
               Err(format!("HTTP {}", status))
           }
           Err(e) => {
               tracing::error!("[relay] Connection to {} failed: {}", destination, e);
               // Re-enqueue
               for msg in messages {
                   let _ = self.persistence.enqueue_offline(destination, &msg);
               }
               let mut backoffs = self.backoff_states.lock().unwrap();
               backoffs.entry(destination.to_string())
                   .or_insert_with(BackoffState::new)
                   .record_failure();
               Err(format!("Connection error: {}", e))
           }
       }
   }
   ```

// Start background cleanup task (D-12: hourly dedup + queue retention cleanup)
let cleanup_persistence = persistence.clone();
let _cleanup_handle = tokio::spawn(async move {
    let mut interval = tokio::time::interval(std::time::Duration::from_secs(3600));
    loop {
        interval.tick().await;
        // Clean dedup cache: remove entries older than 10 minutes (generous beyond 5-min window)
        match cleanup_persistence.cleanup_expired_dedup(chrono::Duration::minutes(10)) {
            Ok(n) if n > 0 => tracing::info!("[cleanup] Removed {} expired dedup entries", n),
            Err(e) => tracing::error!("[cleanup] Dedup cleanup failed: {}", e),
            _ => {}
        }
        // Clean offline queue: remove entries older than 7 days (D-12: 7-day retention)
        match cleanup_persistence.cleanup_expired_queue(chrono::Duration::days(7)) {
            Ok(n) if n > 0 => tracing::info!("[cleanup] Removed {} expired queue entries", n),
            Err(e) => tracing::error!("[cleanup] Queue cleanup failed: {}", e),
            _ => {}
        }
    }
});
