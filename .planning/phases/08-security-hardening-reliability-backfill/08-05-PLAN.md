```bash
#!/usr/bin/env bash
# DEMO-03: Partition Healing
#
# Proves that when a remote server goes offline, messages are persisted
# to disk and automatically delivered when it comes back online.
#
# Scenario:
#   1. Start both servers (server-a human, server-b agent)
#   2. Send a message from server-a to verify federation works
#   3. Stop server-b (simulating partition)
#   4. Send messages from server-a (they queue in server-a's persistent queue)
#   5. Restart server-b
#   6. Verify queued messages are delivered to server-b
#
# Prerequisites:
#   - Docker and docker-compose installed
#   - Run from the demo/ directory
#
# Manual alternative (no Docker):
#   Terminal 1: cargo run -- --mode human --identity server-a.local --cert certs/server-a.pem --key certs/server-a-key.pem --ca-cert certs/ca.pem --federation-port 8443 --remote server-b.local:9443 --data-dir ./data-a
#   Terminal 2: cargo run -- --mode agent --identity server-b.local --cert certs/server-b.pem --key certs/server-b-key.pem --ca-cert certs/ca.pem --federation-port 9443 --remote server-a.local:8443 --port 9080 --data-dir ./data-b
#   Then Ctrl+C terminal 2, send messages in terminal 1, restart terminal 2.

set -euo pipefail
SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
DEMO_DIR="$(dirname "$SCRIPT_DIR")"
cd "$DEMO_DIR"

echo "========================================"
echo "  DEMO-03: Partition Healing"
echo "========================================"
echo ""

# Step 1: Build and start both servers
echo "[Step 1] Starting both servers..."
docker-compose up -d --build
echo "Waiting 5s for servers to initialize..."
sleep 5

# Step 2: Verify federation works (check logs for successful relay)
echo ""
echo "[Step 2] Verifying federation is working..."
echo "Check server-a logs for relay activity:"
docker-compose logs server-a 2>&1 | tail -5
echo ""

# Step 3: Stop server-b (simulate partition)
echo "[Step 3] Stopping server-b (simulating network partition)..."
docker-compose stop server-b
echo "server-b stopped."
echo ""

# Step 4: Send messages (they will queue)
echo "[Step 4] Messages sent while server-b is down will queue in server-a's persistent store."
echo "In the server-a REPL, type a message and press Enter."
echo "The message will be enqueued (check server-a logs for '[relay] Enqueued message for server-b.local (persistent)')."
echo ""
echo "Press Enter when you have sent at least one message..."
read -r

# Step 5: Restart server-b
echo "[Step 5] Restarting server-b..."
docker-compose start server-b
echo "Waiting 10s for server-b to initialize and queued messages to drain..."
sleep 10

# Step 6: Verify delivery
echo ""
echo "[Step 6] Checking server-b received the queued messages..."
echo "server-b logs (look for received messages):"
docker-compose logs --tail=20 server-b 2>&1 | grep -E "agent.*Received|relay.*Successfully" || echo "(no matching log lines yet)"
echo ""
echo "server-a logs (look for successful relay):"
docker-compose logs --tail=20 server-a 2>&1 | grep -E "relay.*Successfully" || echo "(no matching log lines yet)"
echo ""

echo "========================================"
echo "  DEMO-03 Complete"
echo "========================================"
echo ""
echo "Expected results:"
echo "  - server-a logs show: '[relay] Enqueued message for server-b.local (persistent)'"
echo "  - After restart, server-a logs show: '[relay] Successfully relayed to server-b.local'"
echo "  - server-b logs show the agent received and responded to the queued message"
echo ""
echo "To clean up: docker-compose down -v"
```

```bash
# --- Certificate Rotation (for DEMO-04) ---
# Generate a rotated certificate for server-b with a different key
# This simulates a certificate rotation scenario.
if [ "${1:-}" = "rotate-server-b" ]; then
    echo "Generating rotated certificate for server-b..."
    openssl req -new -x509 -days 365 \
        -subj "/CN=server-b.local" \
        -addext "subjectAltName=DNS:server-b.local,DNS:server-b" \
        -CA ca.pem -CAkey ca-key.pem \
        -out server-b-rotated.pem \
        -keyout server-b-rotated-key.pem \
        -noenc 2>/dev/null
    echo "Generated: server-b-rotated.pem, server-b-rotated-key.pem"
    echo "Fingerprint (new):"
    openssl x509 -in server-b-rotated.pem -outform DER | sha256sum | awk '{print "sha256:" $1}'
    echo "Fingerprint (current):"
    openssl x509 -in server-b.pem -outform DER | sha256sum | awk '{print "sha256:" $1}'
fi
```

**demo/scripts/demo-cert-rotation.sh:**

```bash
#!/usr/bin/env bash
# DEMO-04: Certificate Rotation
#
# Proves that when a server's TLS certificate is rotated, federation
# continues without downtime because the tls_fingerprints array
# supports rotation overlap and TOFU auto-updates the pin.
#
# Scenario:
#   1. Start both servers and verify federation works
#   2. Generate a new certificate for server-b
#   3. Compute fingerprints for both old and new certs
#   4. Update server-b's code/config to serve tls_fingerprints: [new, old]
#   5. Restart server-b with the new certificate
#   6. Send a message and verify federation still works
#   7. Check logs for TOFU auto-update
#
# Prerequisites:
#   - Docker and docker-compose installed
#   - OpenSSL installed
#   - Run from the demo/ directory
#
# Manual alternative (no Docker):
#   1. Start both servers as in DEMO-03 manual instructions
#   2. cd certs && bash generate.sh rotate-server-b
#   3. Note the two fingerprints printed
#   4. Stop server-b (Ctrl+C)
#   5. Restart server-b with --cert certs/server-b-rotated.pem --key certs/server-b-rotated-key.pem
#   6. The .well-known endpoint will serve the new cert's fingerprint
#   7. server-a's TOFU check will see the new fingerprint in tls_fingerprints
#      and auto-update its pin

set -euo pipefail
SCRIPT_DIR="$(cd "$(dirname "$0")" && pwd)"
DEMO_DIR="$(dirname "$SCRIPT_DIR")"
cd "$DEMO_DIR"

echo "========================================"
echo "  DEMO-04: Certificate Rotation"
echo "========================================"
echo ""

# Step 1: Start both servers
echo "[Step 1] Starting both servers..."
docker-compose up -d --build
echo "Waiting 5s for servers to initialize..."
sleep 5

# Step 2: Verify federation works
echo ""
echo "[Step 2] Verifying federation works before rotation..."
echo "server-a logs:"
docker-compose logs --tail=5 server-a 2>&1 | grep -E "federation|relay" || echo "(checking...)"
echo ""

# Step 3: Generate rotated certificate
echo "[Step 3] Generating rotated certificate for server-b..."
cd certs
bash generate.sh rotate-server-b
cd "$DEMO_DIR"
echo ""

# Step 4: Compute fingerprints
echo "[Step 4] Certificate fingerprints:"
OLD_FP=$(openssl x509 -in certs/server-b.pem -outform DER 2>/dev/null | shasum -a 256 | awk '{print "sha256:" $1}')
NEW_FP=$(openssl x509 -in certs/server-b-rotated.pem -outform DER 2>/dev/null | shasum -a 256 | awk '{print "sha256:" $1}')
echo "  Current (old): $OLD_FP"
echo "  Rotated (new): $NEW_FP"
echo ""

# Step 5: Swap certificate and restart both servers
echo "[Step 5] Swapping server-b certificate and restarting..."
cp certs/server-b.pem certs/server-b-backup.pem
cp certs/server-b-key.pem certs/server-b-key-backup.pem
cp certs/server-b-rotated.pem certs/server-b.pem
cp certs/server-b-rotated-key.pem certs/server-b-key.pem
echo "Certificate swapped. Restarting both servers to pick up new cert and re-resolve identities..."
docker-compose restart
echo "Waiting 5s for servers to restart..."
sleep 5

# Step 6: Verify federation continues
echo ""
echo "[Step 6] Verifying federation continues after cert rotation..."
echo "Send a message in the server-a REPL to test federation."
echo "Press Enter after sending a message..."
read -r

echo "server-a logs (look for TOFU pinning and successful relay):"
docker-compose logs --tail=20 server-a 2>&1 | grep -E "tofu|relay.*Success" || echo "(no matching log lines yet)"
echo ""
echo "server-b logs:"
docker-compose logs --tail=10 server-b 2>&1 | grep -E "agent.*Received|federation" || echo "(checking...)"
echo ""

# Step 7: Restore original certificate
echo "[Step 7] Restoring original server-b certificate..."
cp certs/server-b-backup.pem certs/server-b.pem
cp certs/server-b-key-backup.pem certs/server-b-key.pem
rm -f certs/server-b-backup.pem certs/server-b-key-backup.pem
rm -f certs/server-b-rotated.pem certs/server-b-rotated-key.pem
echo "Original certificate restored."
echo ""

echo "========================================"
echo "  DEMO-04 Complete"
echo "========================================"
echo ""
echo "Expected results:"
echo "  - server-a logs show: '[tofu] First contact -- pinned server-b.local -> sha256:...'"
echo "  - After rotation, federation relay succeeds (server-a relays to server-b)"
echo "  - server-b agent receives and responds to the message"
echo ""
echo "Key mechanism:"
echo "  - server-b restarts with new cert, serves new fingerprint in .well-known"
echo "  - server-a restarts and re-resolves server-b identity (gets new fingerprint)"
echo "  - TOFU pins the new fingerprint on first contact after restart"
echo ""
echo "To clean up: docker-compose down -v"
```
