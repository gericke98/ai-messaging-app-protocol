    ```rust
    //! AgentRegistry tests — closes RAT-04 (Phase 15).
    //!
    //! Covers:
    //!  - happy path register/cleanup/broadcast
    //!  - CONCURRENT register on same identity (multi-instance — both channels receive)
    //!  - CONCURRENT cleanup + register (no panic; consistent post-state)
    //!  - send_to_agent broadcast to N channels
    //!  - agent_cards lifecycle (entry removed when last channel cleaned up)
    //!  - send_to_agent on unregistered identity (no-op, no panic)
    //!  - register-overwrite of agent_card mid-session (visibility test for Phase 16 Open Q §3)
    //!
    //! Note: tests use `tokio::join!` / `tokio::spawn` — single sequential awaits
    //! do NOT exercise the race conditions (see RESEARCH.md Pitfall 6).

    use ainm_demo::server::AgentRegistry;
    use serde_json::json;
    use std::time::Duration;

    fn card(name: &str) -> serde_json::Value {
        json!({ "name": name, "version": "0.1.0" })
    }

    #[tokio::test]
    async fn register_then_cleanup_happy_path() {
        let reg = AgentRegistry::new();
        let _rx = reg.register("alice@x.local", &card("alice")).await;
        assert!(reg.list_agents().await.contains(&"alice@x.local".to_string()));
        reg.cleanup("alice@x.local").await;
        assert!(!reg.list_agents().await.contains(&"alice@x.local".to_string()),
                "after cleanup, identity should be absent from list_agents()");
    }

    #[tokio::test]
    async fn concurrent_register_same_identity_keeps_both_channels() {
        let reg = AgentRegistry::new();
        let card = card("alice");

        // Two concurrent registers on the same identity.
        let (mut rx1, mut rx2) = tokio::join!(
            reg.register("alice@x.local", &card),
            reg.register("alice@x.local", &card),
        );

        // Both channels are alive — broadcast hits both.
        reg.send_to_agent("alice@x.local", "hello").await;

        let m1 = tokio::time::timeout(Duration::from_millis(100), rx1.recv()).await;
        let m2 = tokio::time::timeout(Duration::from_millis(100), rx2.recv()).await;
        assert_eq!(m1.unwrap().unwrap(), "hello", "rx1 must receive broadcast");
        assert_eq!(m2.unwrap().unwrap(), "hello", "rx2 must receive broadcast");
    }

    #[tokio::test]
    async fn cleanup_removes_all_channels_for_identity() {
        let reg = AgentRegistry::new();
        let _rx1 = reg.register("alice@x.local", &card("alice")).await;
        let _rx2 = reg.register("alice@x.local", &card("alice")).await;

        reg.cleanup("alice@x.local").await;

        let agents = reg.list_agents().await;
        assert!(!agents.contains(&"alice@x.local".to_string()),
                "cleanup should remove the identity entirely (current behavior)");
        // After cleanup, send_to_agent is a no-op (not a panic).
        reg.send_to_agent("alice@x.local", "post-cleanup").await;
    }

    #[tokio::test]
    async fn concurrent_cleanup_and_register_no_panic() {
        // Note: requires AgentRegistry: Clone (or wrap in Arc) — see Task 1.
        let reg = AgentRegistry::new();
        let _rx_initial = reg.register("bob@x.local", &card("bob")).await;

        let reg2 = reg.clone();
        let cleanup_task = tokio::spawn(async move {
            reg2.cleanup("bob@x.local").await;
        });
        let reg3 = reg.clone();
        let register_task = tokio::spawn(async move {
            reg3.register("bob@x.local", &card("bob")).await
        });

        // Both must complete without panic.
        let (cleanup_res, reg_res) = tokio::join!(cleanup_task, register_task);
        assert!(cleanup_res.is_ok(), "cleanup task panicked");
        assert!(reg_res.is_ok(), "register task panicked");

        // Final state: 0 or 1 channels, depending on schedule. Either is acceptable;
        // the invariant being tested is "no panic / no deadlock / no corrupted internal state".
        let _agents = reg.list_agents().await;  // must complete without hanging
    }

    #[tokio::test]
    async fn send_to_unregistered_identity_is_noop() {
        let reg = AgentRegistry::new();
        // Must not panic, must not deadlock.
        reg.send_to_agent("ghost@x.local", "anyone there?").await;
        assert!(!reg.list_agents().await.contains(&"ghost@x.local".to_string()));
    }

    #[tokio::test]
    async fn agent_card_overwrite_is_observable() {
        // Phase 16 Open Question §3 — register currently overwrites the agent_card map
        // for an existing identity. Phase 16 will gate this; Phase 15 just makes it
        // observable in tests so a regression / behavior change is caught.
        let reg = AgentRegistry::new();
        let _rx1 = reg.register("alice@x.local", &card("alice-v1")).await;
        let _rx2 = reg.register("alice@x.local", &card("alice-v2")).await;

        // The registry remains a single-identity entry (multi-channel) — list_agents
        // returns it once.
        let agents = reg.list_agents().await;
        let alice_count = agents.iter().filter(|a| a.as_str() == "alice@x.local").count();
        assert_eq!(alice_count, 1, "list_agents should de-dupe by identity");

        // (We intentionally do NOT assert what agent_card v1 vs v2 wins, since the
        // current API does not expose agent_cards — Phase 16 will add that surface
        // and this test will be extended.)
    }

    #[tokio::test]
    async fn dropped_receiver_eventually_cleaned_up_or_silently_ignored() {
        // When a receiver is dropped (simulating a closed websocket), send_to_agent
        // must not panic. The channel send may fail silently — verify the registry
        // still functions for OTHER identities.
        let reg = AgentRegistry::new();
        let rx_drop = reg.register("alice@x.local", &card("alice")).await;
        let mut rx_keep = reg.register("bob@x.local", &card("bob")).await;
        drop(rx_drop);

        // Sending to alice (dead receiver) must not poison the registry.
        reg.send_to_agent("alice@x.local", "into-the-void").await;

        // Bob must still receive normally.
        reg.send_to_agent("bob@x.local", "still-here").await;
        let m = tokio::time::timeout(Duration::from_millis(100), rx_keep.recv()).await;
        assert_eq!(m.unwrap().unwrap(), "still-here",
                   "registry must remain functional after a dropped receiver on a different identity");
    }
    ```
