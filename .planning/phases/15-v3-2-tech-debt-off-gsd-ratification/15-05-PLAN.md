    **Step 1 — Create `packages/connector/src/log.ts`:**

    A 30-50 line module that exports a `log` object with `info`, `warn`, `error`, `debug` methods. Each method:
    - Accepts an optional context object (`{ err?, apiKey?, ... }`) and a message string
    - Calls the matching `console.*` method, but FIRST scrubs the context object via `redact()`
    - The `redact()` function recursively walks the object and replaces values for known sensitive keys with `'[REDACTED]'`

    Sensitive key list (case-INsensitive):
    - `apiKey`, `api_key`
    - `authorization` (HTTP header capture)
    - `briefing` (compound memory payload)
    - `signature` (Ed25519 signature)
    - `privateKey`, `private_key`
    - `seed` (key seed)
    - `token`, `accessToken`, `refreshToken`
    - `OPENAI_API_KEY`, `ANTHROPIC_API_KEY` (literal env var keys when caller passes process.env)

    Implementation skeleton:
    ```typescript
    // packages/connector/src/log.ts
    // Redacted structured logger for AINM connector SDK (RAT-01 row 4 — Phase 15-05).
    // Hand-rolled per RESEARCH.md Open Q §4 — pino is overkill for ratification scope.

    const SENSITIVE_KEYS = new Set([
      "apikey", "api_key",
      "authorization",
      "briefing",
      "signature",
      "privatekey", "private_key",
      "seed",
      "token", "accesstoken", "refreshtoken",
      "openai_api_key", "anthropic_api_key",
    ]);

    function isSensitive(key: string): boolean {
      return SENSITIVE_KEYS.has(key.toLowerCase());
    }

    export function redact(value: unknown, depth = 0): unknown {
      if (depth > 6) return "[TRUNCATED]";
      if (value === null || typeof value !== "object") return value;
      if (Array.isArray(value)) return value.map((v) => redact(v, depth + 1));
      const out: Record<string, unknown> = {};
      for (const [k, v] of Object.entries(value as Record<string, unknown>)) {
        out[k] = isSensitive(k) ? "[REDACTED]" : redact(v, depth + 1);
      }
      return out;
    }

    function emit(level: "log" | "info" | "warn" | "error" | "debug", ctx: object | undefined, msg: string): void {
      const safe = ctx === undefined ? "" : JSON.stringify(redact(ctx));
      // eslint-disable-next-line no-console -- single allowed console call site
      (console[level] as (...args: unknown[]) => void)(`[ainm-connector] ${msg}`, safe);
    }

    export const log = {
      debug: (ctx: object | undefined, msg: string) => emit("debug", ctx, msg),
      info:  (ctx: object | undefined, msg: string) => emit("info",  ctx, msg),
      warn:  (ctx: object | undefined, msg: string) => emit("warn",  ctx, msg),
      error: (ctx: object | undefined, msg: string) => emit("error", ctx, msg),
    };
    ```
