```typescript
async function killProcessGroup(proc: ChildProcess, graceMs = 5000): Promise<void> {
  if (!proc.pid || proc.killed) return;
  try {
    process.kill(-proc.pid, "SIGTERM");          // negative pid = process group
  } catch (err) {
    if ((err as NodeJS.ErrnoException).code === "ESRCH") return;
    throw err;
  }
  await new Promise<void>((resolve) => {
    let resolved = false;
    const t = setTimeout(() => {
      if (resolved) return;
      try { process.kill(-proc.pid!, "SIGKILL"); } catch { /* gone */ }
      resolved = true;
      resolve();
    }, graceMs);
    proc.once("exit", () => {
      clearTimeout(t);
      if (!resolved) { resolved = true; resolve(); }
    });
  });
}
```

    ```typescript
    import { describe, it, expect } from "vitest";
    import { spawn, ChildProcess } from "node:child_process";
    import { killProcessGroup } from "../src/claude-code-agent.js";

    function spawnDummyChild(): ChildProcess {
      // Use node -e to spawn a child that itself spawns a grandchild, simulating
      // claude's subprocess tree. The grandchild keeps running so we can detect leaks.
      const code = `
        const { spawn } = require('child_process');
        const grandchild = spawn(process.execPath, ['-e', 'setInterval(()=>{}, 1000)'], { stdio: 'ignore' });
        console.log(grandchild.pid);
        // Keep parent alive too.
        setInterval(() => {}, 1000);
      `;
      return spawn(process.execPath, ["-e", code], {
        detached: true,
        stdio: ["pipe", "pipe", "pipe"],
      });
    }

    function isProcessAlive(pid: number): boolean {
      try {
        process.kill(pid, 0);   // signal 0 = check existence, no-op if alive
        return true;
      } catch (err) {
        if ((err as NodeJS.ErrnoException).code === "ESRCH") return false;
        throw err;
      }
    }

    describe("Process cleanup (RAT-02 — leak detection)", () => {
      it("kills the direct child process", async () => {
        const proc = spawnDummyChild();
        expect(proc.pid).toBeDefined();
        const childPid = proc.pid!;
        await killProcessGroup(proc, 2000);
        // Allow OS a moment to reap.
        await new Promise((r) => setTimeout(r, 100));
        expect(isProcessAlive(childPid)).toBe(false);
      });

      it("kills the entire process group (grandchild also gone)", async () => {
        const proc = spawnDummyChild();
        // Read first line of stdout to get grandchild pid.
        const grandchildPid = await new Promise<number>((resolve, reject) => {
          let buf = "";
          proc.stdout?.on("data", (d) => {
            buf += d.toString();
            const line = buf.split("\n")[0];
            if (line) {
              const pid = parseInt(line.trim(), 10);
              if (!isNaN(pid)) resolve(pid);
            }
          });
          setTimeout(() => reject(new Error("grandchild pid not received")), 3000);
        });

        expect(isProcessAlive(grandchildPid)).toBe(true);
        await killProcessGroup(proc, 2000);
        await new Promise((r) => setTimeout(r, 200));
        expect(isProcessAlive(grandchildPid)).toBe(false);
      });

      it("escalates to SIGKILL when SIGTERM is ignored", async () => {
        // Spawn a child that ignores SIGTERM.
        const proc = spawn(process.execPath, ["-e", `
          process.on('SIGTERM', () => { /* swallow */ });
          setInterval(()=>{}, 1000);
        `], { detached: true, stdio: "ignore" });
        const pid = proc.pid!;
        const start = Date.now();
        await killProcessGroup(proc, 500);    // short grace so test runs fast
        const elapsed = Date.now() - start;
        await new Promise((r) => setTimeout(r, 100));
        expect(isProcessAlive(pid)).toBe(false);
        // Must have waited at least the grace period before SIGKILL.
        expect(elapsed).toBeGreaterThanOrEqual(500);
        // But not too long (sanity bound).
        expect(elapsed).toBeLessThan(2000);
      });

      it("is a no-op for already-exited process", async () => {
        const proc = spawn(process.execPath, ["-e", "process.exit(0)"], { detached: true });
        await new Promise((r) => proc.once("exit", r));
        // Should NOT throw.
        await killProcessGroup(proc, 100);
      });
    });
    ```

    ```markdown
    ## D-02: `--bare` mode vs OAuth-mode tradeoff
    **Decision:** The connector currently REQUIRES `--bare` mode (API key via `ANTHROPIC_API_KEY` env var). OAuth-mode Claude Code (browser-based login flow used by interactive humans) is NOT supported in v0.1.x.

    **Rationale:** OAuth-mode requires a browser-interactive flow that's incompatible with headless connector deployments (server-spawned, CI, container). `--bare` mode delegates auth to a long-lived API key the operator manages out-of-band.

    **Tradeoffs:**
    - **`--bare` pros:** Deterministic; no browser dependency; works in CI/container; key rotation is operator-managed.
    - **`--bare` cons:** API key never expires unless rotated explicitly; key leak = full account exposure; no per-session auth scope.
    ```
