    ```swift
    import XCTest

    /// Base class for all AINM UI tests. Handles app launch with seed + optional event script,
    /// waits for the `app.ready` marker rendered by AINMMessengerApp once bootstrap completes,
    /// and cleans up on teardown.
    class AINMUITestCase: XCTestCase {

        var app: XCUIApplication!

        override func setUpWithError() throws {
            try super.setUpWithError()
            continueAfterFailure = false
            app = XCUIApplication()
        }

        override func tearDownWithError() throws {
            app?.terminate()
            try super.tearDownWithError()
        }

        /// Launch the app with `-UITestSeed <fixture>` and optional `-UITestEvents <script>`.
        /// Blocks until the `app.ready` marker appears (bootstrap complete).
        /// Fails the test with a clear message if bootstrap doesn't complete within `timeout` seconds.
        func launchWith(seedFixture: String,
                        eventScript: String? = nil,
                        timeout: TimeInterval = 10,
                        file: StaticString = #file,
                        line: UInt = #line) {
            var args = ["-UITestSeed", seedFixture]
            if let eventScript {
                args += ["-UITestEvents", eventScript]
            }
            app.launchArguments = args
            app.launch()

            let ready = app.otherElements[UITestID.App.ready]
            XCTAssertTrue(ready.waitForExistence(timeout: timeout),
                          "App never signalled ready (UITestID.App.ready) after seed load within \(timeout)s. " +
                          "Check tmp/uat-server.log for server issues or the test runner console for bootstrap errors.",
                          file: file, line: line)
        }

        /// Convenience: navigate into a seeded thread via its conversation list cell.
        /// Precondition: launchWith has already been called and returned.
        @discardableResult
        func openThread(threadId: String,
                        timeout: TimeInterval = 5,
                        file: StaticString = #file,
                        line: UInt = #line) -> Bool {
            let cellId = UITestID.ConversationList.cell(threadId: threadId)
            // Conversation list cells can be buttons or other elements depending on SwiftUI version.
            // Try both; first match wins.
            let button = app.buttons[cellId]
            let other = app.otherElements[cellId]
            if button.waitForExistence(timeout: timeout) {
                button.tap()
                return true
            }
            if other.waitForExistence(timeout: 1) {
                other.tap()
                return true
            }
            XCTFail("Conversation list cell for threadId=\(threadId) not found within \(timeout)s",
                    file: file, line: line)
            return false
        }
    }
    ```

Error and timedOut states are covered by the existence of the icon ids alone — `UITestID.ToolCall.statusError` and `statusTimedOut` are declared and present in the view code from 14-05; dedicated tests for error/timeout would need separate event scripts (deferred — scenario 1 test verifies the one end-to-end lifecycle).

    ```swift
    import XCTest

    final class ToolCallIndicatorUITests: AINMUITestCase {

        func test_toolCallLifecycle_runningTransitionsToCompleted_withExpandableDetail() {
            launchWith(seedFixture: "scenario-01-tool-call",
                       eventScript: "scenario-01-events")

            openThread(threadId: "thread-01")

            // 0ms: started → running spinner appears
            let running = app.images[UITestID.ToolCall.statusRunning]
                .firstMatch
            XCTAssertTrue(running.waitForExistence(timeout: 3),
                          "Running spinner did not appear after tool.call.started")

            // 2000ms: completed → green checkmark replaces spinner
            let completed = app.images[UITestID.ToolCall.statusCompleted]
                .firstMatch
            XCTAssertTrue(completed.waitForExistence(timeout: 6),
                          "Completed checkmark did not appear after tool.call.completed (2s+margin)")

            // After transition, the running spinner must be removed from the hierarchy —
            // per spec/tool-visibility.md, only one lifecycle state is visible at a time.
            XCTAssertFalse(running.exists,
                           "Running spinner should be gone after tool.call.completed lifecycle event")

            // Expand the indicator → input JSON visible
            let expand = app.buttons[UITestID.ToolCall.expandButton]
                .firstMatch
            if !expand.waitForExistence(timeout: 2) {
                // Fall back: try images collection if the chevron is an Image not a Button.
                let expandImage = app.images[UITestID.ToolCall.expandButton].firstMatch
                XCTAssertTrue(expandImage.waitForExistence(timeout: 2),
                              "Expand chevron not queryable via buttons or images")
                expandImage.tap()
            } else {
                expand.tap()
            }

            // Detail view sections appear
            let inputJson = app.staticTexts[UITestID.ToolCall.detailInputJson]
                .firstMatch
            XCTAssertTrue(inputJson.waitForExistence(timeout: 2),
                          "Expanded detail did not show input JSON")

            let result = app.staticTexts[UITestID.ToolCall.detailResult]
                .firstMatch
            XCTAssertTrue(result.waitForExistence(timeout: 2),
                          "Expanded detail did not show result summary")
        }

        func test_identifiers_forAllFourStates_exist_in_UITestID_namespace() {
            // Compile-time guard: ensures UITestID exposes all 4 state ids, so a fixture that fires
            // error or timedOut events would be queryable without code changes.

            _ = UITestID.ToolCall.statusCompleted
            _ = UITestID.ToolCall.statusError
            _ = UITestID.ToolCall.statusTimedOut
        }
    }
    ```
