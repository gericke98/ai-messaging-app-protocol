    **Critical:** Use the EXACT 32 bytes shown (`UAT_USER@LOCALHOST` + 0xDEADBEEFFEEDFACE marker) so that `uat-user@localhost`'s public key is stable across all six fixtures and CI runs. Do not regenerate.

    ```swift
    #if DEBUG
    import CryptoKit
    import Foundation

    /// Installs a deterministic Ed25519 keypair for the `uat-user@localhost` identity
    /// so every UAT run starts from the same cryptographic state.
    ///
    /// Mirrors `KeyManager`'s Keychain contract (service="com.ainm.messenger",
    /// account="ed25519.privateKey") — after `resetIdentity()` runs, the next call to
    /// `KeyManager().getOrCreateKeypair()` loads this keypair instead of generating a new one.
    enum UITestIdentityBootstrap {

        /// 32-byte deterministic seed for the `uat-user@localhost` private key.
        /// Layout: ASCII "UAT_USER@LOCALHOST" + zero padding + magic marker.
        /// Do NOT regenerate — fixture signatures and cross-run reproducibility depend on this.
        static let uatSeedBytes: [UInt8] = [
            0x55, 0x41, 0x54, 0x5F, 0x55, 0x53, 0x45, 0x52, // "UAT_USER"
            0x40, 0x4C, 0x4F, 0x43, 0x41, 0x4C, 0x48, 0x4F, // "@LOCALHO"
            0x53, 0x54, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, // "ST" + zero padding
            0xDE, 0xAD, 0xBE, 0xEF, 0xFE, 0xED, 0xFA, 0xCE  // magic marker
        ]

        /// Logical identity address that the seeded keypair represents.
        /// Fixture JSON files set `identity: "uat-user@localhost"` to match.
        static let identityAddress = "uat-user@localhost"

        enum Error: Swift.Error {
            case seedSizeInvalid(Int)
        }

        /// Deletes any existing keypair and installs the deterministic one.
        /// Idempotent — calling multiple times produces identical results.
        static func resetIdentity() async throws {
            guard uatSeedBytes.count == 32 else {
                throw Error.seedSizeInvalid(uatSeedBytes.count)
            }

            // Step 1: remove any existing keypair. KeyManager.deleteKeypair() is already
            // a no-op if the Keychain item isn't present (errSecItemNotFound handled).
            let manager = KeyManager()
            try await manager.deleteKeypair()

            // Step 2: construct the deterministic private key.
            let seed = Data(uatSeedBytes)
            let privateKey = try Curve25519.Signing.PrivateKey(rawRepresentation: seed)

            // Step 3: persist via the same Keychain path KeyManager uses for load.
            // If KeychainStorage is not directly accessible from this module (e.g. `fileprivate`),
            // replace with KeyManager's public `loadOrCreateWithSeed(_:)` or add a new
            // `installPrivateKey(_:)` public helper on KeyManager and call it here.
            let storage = KeyManager.KeychainStorage()
            try storage.save(privateKey.rawRepresentation)
        }

        /// Deterministic public key for assertions. Useful for tests that want to
        /// verify signatures against a known value without recomputing from bytes.
        static var deterministicPublicKeyBase64: String {
            let seed = Data(uatSeedBytes)
            // Force-try is safe here: 32 valid bytes always produce a valid private key.
            let privateKey = try! Curve25519.Signing.PrivateKey(rawRepresentation: seed)
            return privateKey.publicKey.rawRepresentation.base64EncodedString()
        }
    }
    #endif
    ```

    **File B: `ios/AINMMessenger/AINMMessenger/Testing/SeedLoader.swift`** — the DB wipe + transactional insert.

    ```swift
    #if DEBUG
    import Foundation
    import GRDB

    enum SeedLoader {

        enum Error: Swift.Error, CustomStringConvertible {
            case fixtureNotFound(String)
            case decodeFailed(String, underlying: any Swift.Error)
            case databaseNotReady

            var description: String {
                switch self {
                case .fixtureNotFound(let name): return "Fixture not found in app bundle: \(name).json"
                case .decodeFailed(let name, let err): return "Failed to decode fixture \(name): \(err)"
                case .databaseNotReady: return "AppDatabase not initialized before fixture load"
                }
            }
        }

        /// MUST run BEFORE AppDatabase.makeDefault() to avoid ValueObservation flicker
        /// (Pitfall 6). Deletes the SQLite main + WAL + SHM files.
        static func preparePristineDatabase() throws {
            let fm = FileManager.default
            let appSupport = try fm.url(for: .applicationSupportDirectory,
                                        in: .userDomainMask,
                                        appropriateFor: nil,
                                        create: true)
            let dbURL = appSupport.appendingPathComponent("ainm.sqlite")
            for suffix in ["", "-wal", "-shm"] {
                let url = suffix.isEmpty
                    ? dbURL
                    : URL(fileURLWithPath: dbURL.path + suffix)
                try? fm.removeItem(at: url)
            }
        }

        /// Loads a fixture JSON from the app bundle (NOT the test bundle — SeedLoader runs
        /// in the app target; fixtures ship with the app via the synchronized root group —
        /// see RESEARCH.md "Bundle handling for fixtures").
        /// Writes all rows in a single transaction so ValueObservation sees one atomic change.
        @MainActor
        static func loadFixtureIfRequested(fixture name: String, into appState: AppState) async throws {
            guard let url = Bundle.main.url(forResource: name, withExtension: "json") else {
                throw Error.fixtureNotFound(name)
            }
            let data = try Data(contentsOf: url)
            let fixture: ScenarioFixture
            do {
                fixture = try JSONDecoder().decode(ScenarioFixture.self, from: data)
            } catch {
                throw Error.decodeFailed(name, underlying: error)
            }

            guard let dbPool = appState.appDatabase?.dbPool else {
                throw Error.databaseNotReady
            }

            try await dbPool.write { db in
                for t in fixture.threads       { try t.asRecord().insert(db) }
                for p in fixture.participants  { try p.asRecord().insert(db) }
                for m in fixture.messages      { try m.asRecord().insert(db) }
                for tc in fixture.toolCalls    { try tc.asRecord().insert(db) }
                for af in fixture.archivalFacts { try af.asRecord().insert(db) }
                for ac in fixture.agentCards   { try ac.asRecord().insert(db) }
            }
        }
    }
    ```
