    async fn run_uat(_cli: Cli) -> Result<()> {
        use tracing::Level;

        // 1. Logging: INFO level, file writer at tmp/uat-server.log. Repo-relative path —
        //    Makefile cd's into demo/ then `../tmp/uat-server.log` is the target. run_uat
        //    normalizes both working directories by resolving relative to CARGO_MANIFEST_DIR's parent.
        let repo_root = std::path::Path::new(env!("CARGO_MANIFEST_DIR"))
            .parent()
            .expect("CARGO_MANIFEST_DIR has parent");
        let log_dir = repo_root.join("tmp");
        std::fs::create_dir_all(&log_dir)?;
        let log_path = log_dir.join("uat-server.log");
        let log_file = std::fs::File::create(&log_path)?;
        let _ = tracing_subscriber::fmt()
            .with_target(false)
            .with_max_level(Level::INFO)
            .with_writer(std::sync::Mutex::new(log_file))
            .try_init();

        // 2. Ephemeral SQLite: use system tempdir so teardown is automatic.
        //    The legacy demo uses in-memory ServerState; if a DB is actually required by
        //    start_with_port (check demo/src/db.rs), compute a tempdir path and pass it in.
        //    Keep this minimal — UAT mode only needs WebSocket + /health for the iOS client.
        let tmp_dir = tempfile::tempdir()?;
        tracing::info!("[uat] ephemeral db dir: {}", tmp_dir.path().display());

        // 3. Provision the same two identities the existing run_legacy_demo uses:
        //    uat-user@localhost (human) and helper@localhost (agent). If run_legacy_demo
        //    uses different names, mirror IT exactly — the iOS fixtures must match.
        let human = identity::generate_keypair("uat-user@localhost")?;
        let agent = identity::generate_keypair("helper@localhost")?;
        let identity_store = identity::IdentityStore::new();
        identity_store.register(&human.address, &human.public_key_b64(), "human", "localhost", None);
        identity_store.register(
            &agent.address,
            &agent.public_key_b64(),
            "agent",
            "localhost",
            Some("http://localhost:8080/.well-known/ainm/helper/agent-card.json".to_string()),
        );

        // 4. Build the agent card JSON. Reuse run_legacy_demo's agent_card construction —
        //    if it's a local helper fn, lift it or call it. If it's inline, copy the JSON.
        let agent_card = build_uat_agent_card();

        // 5. Start server on fixed port 8080 with /health registered.
        let (addr, handle) = server::start_with_port(identity_store.clone(), agent_card, 8080).await?;
        tracing::info!("[uat] server ready on ws://{}/ws, GET /health returns 200", addr);
        println!("[uat] server ready on ws://{}/ws", addr);   // stdout line helps humans

        // 6. Block on Ctrl-C. Makefile sends SIGTERM on trap EXIT/INT/TERM.
        tokio::signal::ctrl_c().await?;
        tracing::info!("[uat] received ctrl-c; shutting down");
        handle.abort();

        Ok(())
    }

        // If run_legacy_demo already builds one, mirror its shape.
        serde_json::json!({
            "name": "helper",

            "description": "UAT helper agent",
            "capabilities": ["streaming", "tools", "memory", "sub-threads"],
            "skills": [],
            "tools": []
        })
    }
    ```

    # Phase 14 UAT automation harness orchestrator.
    # See docs/testing.md for full usage.

    .PHONY: uat uat-clean uat-debug

    UAT_SERVER_LOG       := tmp/uat-server.log
    UAT_SERVER_PID       := tmp/uat-server.pid
    UAT_PORT             := 8080
    UAT_HEALTH_URL       := http://localhost:$(UAT_PORT)/health
    UAT_HEALTH_TIMEOUT_S := 30
    UAT_SCHEME           := AINMMessengerUITests
    UAT_DESTINATION      := 'platform=iOS Simulator,name=iPhone 16,OS=18.0'
    UAT_RESULT_BUNDLE    := tmp/UAT-Results

    uat: ## Run hermetic UAT suite (boot server, poll health, run xcodebuild test, teardown)
    	@mkdir -p tmp
    	@# Preflight: fail fast if port is in use (Pitfall 8 — tokio bind would otherwise fail silently in background)
    	@if lsof -nP -iTCP:$(UAT_PORT) -sTCP:LISTEN >/dev/null 2>&1; then \
    	  echo "[uat] port $(UAT_PORT) already in use. Run 'lsof -i :$(UAT_PORT)' to identify; aborting." >&2 ; exit 1 ; \
    	fi
    	@# Start server in background; capture PID so the trap can kill it.
    	@( cd demo && cargo run --quiet --bin ainm-demo -- --uat > ../$(UAT_SERVER_LOG) 2>&1 & echo $$! > ../$(UAT_SERVER_PID) )
    	@trap 'echo "[uat] tearing down server (PID $$(cat $(UAT_SERVER_PID) 2>/dev/null))"; kill $$(cat $(UAT_SERVER_PID) 2>/dev/null) 2>/dev/null; rm -f $(UAT_SERVER_PID)' EXIT INT TERM ; \
    	 echo "[uat] waiting for server readiness at $(UAT_HEALTH_URL) (timeout $(UAT_HEALTH_TIMEOUT_S)s)" ; \
    	 for i in $$(seq 1 $(UAT_HEALTH_TIMEOUT_S)) ; do \
    	   if curl -sf -o /dev/null "$(UAT_HEALTH_URL)" ; then \
    	     echo "[uat] server ready after $$i second(s)" ; break ; \
    	   fi ; \
    	   if [ $$i -eq $(UAT_HEALTH_TIMEOUT_S) ] ; then \
    	     echo "[uat] server not ready after $(UAT_HEALTH_TIMEOUT_S)s (tail of $(UAT_SERVER_LOG) below):" >&2 ; \
    	     tail -20 $(UAT_SERVER_LOG) >&2 || true ; \
    	     exit 1 ; \
    	   fi ; \
    	   sleep 1 ; \
    	 done ; \
    	 echo "[uat] running xcodebuild test -scheme $(UAT_SCHEME)..." ; \
    	 ( cd ios/AINMMessenger && xcodebuild test \
    	     -project AINMMessenger.xcodeproj \
    	     -scheme $(UAT_SCHEME) \
    	     -destination $(UAT_DESTINATION) \
    	     CODE_SIGNING_ALLOWED=NO \
    	     -resultBundlePath ../../$(UAT_RESULT_BUNDLE) )

    uat-debug: ## Boot the UAT server and hold it until Ctrl-C (for manual inspection; no tests)
    	@mkdir -p tmp
    	@echo "[uat-debug] booting ainm-demo --uat on port $(UAT_PORT); Ctrl-C to stop"
    	@cd demo && cargo run --bin ainm-demo -- --uat

    uat-clean: ## Remove UAT artifacts (logs, result bundles, PID file)
    	rm -rf tmp/
    ```
